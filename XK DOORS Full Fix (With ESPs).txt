local function SafeLoad(Url, Label)
	local tag = tostring(Label or Url)
	if type(Url) ~= "string" or #Url == 0 then
		warn("[SafeLoad] Geçersiz URL (" .. tag .. ")")
		return nil
	end
	local okFetch, sourceOrErr = pcall(function()
		return game:HttpGet(Url)
	end)
	if not okFetch or type(sourceOrErr) ~= "string" or #sourceOrErr == 0 then
		warn("[SafeLoad] İndirme başarısız (" .. tag .. "): " .. tostring(sourceOrErr))
		return nil
	end
	local fn, compileErr = loadstring(sourceOrErr, "@" .. tag)
	if not fn then
		warn("[SafeLoad] Derleme hatası (" .. tag .. "): " .. tostring(compileErr))
		return nil
	end
	local okRun, result = pcall(fn)
	if not okRun then
		warn("[SafeLoad] Çalıştırma hatası (" .. tag .. "): " .. tostring(result))
		return nil
	end
	return result
end

local CurrentLang = "zh"
local TranslatableElements = {}

local function GetTranslatedText(originalText, lang)
	if lang == "zh" or type(originalText) ~= "string" then
		return originalText
	end
	local entry = Translations[originalText]
	if entry and entry[lang] then
		return entry[lang]
	end
	return originalText
end

local function RegisterTranslatable(setFn, getFn)
	table.insert(TranslatableElements, { set = setFn, get = getFn })
end

local function ApplyLanguage(lang)
	CurrentLang = (lang == "tr" or lang == "en" or lang == "zh") and lang or "tr"

	for _, item in ipairs(TranslatableElements) do
		pcall(function()
			local orig = item.get()
			if type(orig) == "string" then
				item.set(GetTranslatedText(orig, CurrentLang))
			end
		end)
	end

	local root = UILibrary.ScreenGui or Services.CoreGui
	if root then
		for _, desc in ipairs(root:GetDescendants()) do
			if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
				if not desc:GetAttribute("OrigText") then
					desc:SetAttribute("OrigText", desc.Text)
				end
				local origText = desc:GetAttribute("OrigText")
				local mapped = GetTranslatedText(origText, CurrentLang)
				if mapped ~= origText then
					desc.Text = mapped
				elseif CurrentLang == "zh" then
					desc.Text = origText
				end
			end
		end
	end
end

local function WrapTextContainer(container, origText)
	if not container then return end
	if type(container.SetText) == "function" then
		local current = origText
		RegisterTranslatable(
			function(txt) container:SetText(txt) end,
			function() return current end
		)
	end
end

			Lighting.ExposureCompensation = 0.5
		else

			Lighting.ClockTime = OriginalLightingState.ClockTime
			Lighting.FogEnd = OriginalLightingState.FogEnd
			Lighting.GlobalShadows = OriginalLightingState.GlobalShadows

			Lighting.OutdoorAmbient = OriginalLightingState.OutdoorAmbient
			Lighting.ExposureCompensation = OriginalLightingState.ExposureCompensation
		end
	end)
end

local function IsCameraFirstPerson(cam)
	if not cam then return false end
	return (cam.CFrame.Position - cam.Focus.Position).Magnitude <= 1.5
end

local function NeutralizeCameraShake(cam, char)
	if not NoCameraShakeEnabled then return end
	pcall(function()
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.CameraOffset = Vector3.zero
		end
		if cam then
			local shakeVal = cam:FindFirstChild("CameraShake") or cam:FindFirstChild("Shake")
			if shakeVal and (shakeVal:IsA("NumberValue") or shakeVal:IsA("Vector3Value")) then
				if shakeVal:IsA("NumberValue") then
					shakeVal.Value = 0
				elseif shakeVal:IsA("Vector3Value") then
					shakeVal.Value = Vector3.zero
				end
			end
		end
	end)
end

local function DetectCurrentFloorKey()
	local floorName = ""
	pcall(function()
		local rs = Services.ReplicatedStorage
		local gd = rs and rs:FindFirstChild("GameData")
		local fl = gd and gd:FindFirstChild("Floor")
		if fl and fl.Value then
			floorName = tostring(fl.Value)
		end
	end)
	local lower = string.lower(floorName)
	if string.find(lower, "fools", 1, true) or string.find(lower, "april", 1, true) then
		return "AprilFools"
	elseif string.find(lower, "cringle", 1, true) or string.find(lower, "workshop", 1, true) or string.find(lower, "frozen", 1, true) then
		return "CringleWorkshop"
	elseif string.find(lower, "archive", 1, true) then
		return "Archives"
	elseif string.find(lower, "stair", 1, true) then
		return "Stairwell"
	elseif string.find(lower, "outdoor", 1, true) then
		return "OutDoors"
	elseif string.find(lower, "backdoor", 1, true) then
		return "BackDoors"
	elseif string.find(lower, "mine", 1, true) then
		return "Mines"
	elseif string.find(lower, "hotel", 1, true) then
		return "Hotel"
	end
	return nil
end

local function ResolveAliasForCurrentMode(rawKey)
	local entry = Entities[rawKey]
	local alias = entry and entry.Alias or tostring(rawKey)
	local floorKey = DetectCurrentFloorKey()
	if not floorKey then
		return "CustomEntity"
	end
	local allowed = FloorAllowedAliases[floorKey]
	if allowed and not allowed[alias] then
		return "CustomEntity"
	end
	return alias
end

local function IsInsideRoom(inst)
	if not inst or not workspace then return false end

	if not currentRooms then return false end

end

local function IsDropdownItemSelected(dropdownOpt, keyName)
	if not dropdownOpt or type(dropdownOpt.Value) ~= "table" or not keyName then
		return false
	end
	if dropdownOpt.Value[keyName] == true then
		return true
	end
	for _, val in pairs(dropdownOpt.Value) do
		if val == keyName then
			return true
		end
	end
	return false
end

local function IsEntitySelected(aliasName)
	local dd = Options and Options.EntityList
	if not dd or type(dd.Value) ~= "table" then
		return true
	end
	local hasSelected = false
	for _, isOn in pairs(dd.Value) do
		if isOn then
			hasSelected = true
			break
		end
	end
	if not hasSelected then
		return true
	end
	return IsDropdownItemSelected(dd, aliasName) or (IsDropdownItemSelected(dd, "CustomEntity") and aliasName == "CustomEntity")
end

	end)
end

	lbl.TextSize = textSize
	lbl.TextColor3 = color
	lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	lbl.TextTransparency = textTrans
	lbl.TextStrokeTransparency = textStrokeTrans
	lbl.Text = textStr
	lbl.Parent = bb

	ActiveESPObjects[target] = {

		Anchor = anchor,

		BaseText = textStr,

end

local function RegisterTrackedObject(category, obj, label)
	if not obj or typeof(obj) ~= "Instance" then return end
	local bucket = TrackedByCategory[category]
	if not bucket then return end
	if not bucket[obj] then
		bucket[obj] = { Object = obj, Label = label or obj.Name }
	end
end

local function RefreshCategoryESP(category)
	local bucket = TrackedByCategory[category]
	if not bucket then return end

	local toggleMap = {
		Entities = Toggles.EntityESPToggle,
		Structures = Toggles.StructureESPToggle,
		Objectives = Toggles.ObjectiveESPToggle,
		Items = Toggles.ItemESPToggle,
		Players = Toggles.PlayerESPToggle,
		Currency = Toggles.CurrencyESPToggle,
		Doors = Toggles.DoorESPToggle,
		HidingSpots = Toggles.HidingSpotESPToggle,
		Chests = Toggles.ChestESPToggle,
		Ladders = Toggles.LadderESPToggle,
		Misc = Toggles.MiscESPToggle,
	}

	local colorOptMap = {
		Entities = Options.EntityESPColor,
		Structures = Options.StructureESPColor,
		Objectives = Options.ObjectiveESPColor,
		Items = Options.ItemESPColor,
		Players = Options.PlayerESPColor,
		Currency = Options.CurrencyESPColor,
		Doors = Options.DoorESPColor,
		HidingSpots = Options.HidingSpotESPColor,
		Chests = Options.ChestESPColor,
		Ladders = Options.LadderESPColor,
		Misc = Options.MiscESPColor,
	}

	local defaultColors = {
		Entities = Color3.fromRGB(255, 0, 0),
		Structures = Color3.fromRGB(0, 200, 255),
		Objectives = Color3.fromRGB(0, 255, 0),
		Items = Color3.fromRGB(170, 0, 255),
		Players = Color3.fromRGB(255, 255, 255),
		Currency = Color3.fromRGB(255, 255, 0),
		Doors = Color3.fromRGB(0, 200, 255),
		HidingSpots = Color3.fromRGB(255, 170, 0),
		Chests = Color3.fromRGB(255, 255, 0),
		Ladders = Color3.fromRGB(3, 67, 71),
		Misc = Color3.fromRGB(255, 255, 255),
	}

	local tog = toggleMap[category]
	local isEnabled = tog and tog.Value == true
	local colVal = (colorOptMap[category] and colorOptMap[category].Value) or defaultColors[category]

	for obj, info in pairs(bucket) do
		if not obj or not obj.Parent then
			RemoveESP(obj)
			bucket[obj] = nil
		else
			local allowed = isEnabled
			if allowed and category == "Entities" then
				local dd = Options and Options.EntityESPOptions
				if dd and type(dd.Value) == "table" then
					local hasAny = false
					for _, v in pairs(dd.Value) do if v then hasAny = true break end end
					if hasAny then
						allowed = IsDropdownItemSelected(dd, info.Label)
					end
				end
			elseif allowed and category == "Structures" then
				local dd = Options and Options.StructureESPOptions
				if dd and type(dd.Value) == "table" then
					local hasAny = false
					for _, v in pairs(dd.Value) do if v then hasAny = true break end end
					if hasAny then
						allowed = IsDropdownItemSelected(dd, info.Label)
					end
				end
			end

			if allowed then
				AddESP({
					Object = obj,
					Text = info.Label,
					Color = colVal,
					Category = category,
				})
			else
				RemoveESP(obj)
			end
		end
	end
end

local function RefreshAllESPSettings()
	local fillTrans = (Options and Options.ESPFillTransparency and tonumber(Options.ESPFillTransparency.Value)) or 0.75
	local outlineTrans = (Options and Options.ESPOutlineTransparency and tonumber(Options.ESPOutlineTransparency.Value)) or 0
	local textTrans = (Options and Options.ESPTextTransparency and tonumber(Options.ESPTextTransparency.Value)) or 0
	local textStrokeTrans = (Options and Options.ESPTextOutlineTransparency and tonumber(Options.ESPTextOutlineTransparency.Value)) or 0
	local textSize = (Options and Options.ESPTextSize and tonumber(Options.ESPTextSize.Value)) or 16
	local maxDist = (Options and Options.ESPRenderLimit and tonumber(Options.ESPRenderLimit.Value)) or 240

	for _, entry in pairs(ActiveESPObjects) do
		if entry.Highlight then
			entry.Highlight.FillTransparency = fillTrans
			entry.Highlight.OutlineTransparency = outlineTrans
		end
		if entry.Billboard then
			entry.Billboard.MaxDistance = maxDist
		end
		if entry.Label then
			entry.Label.TextTransparency = textTrans
			entry.Label.TextStrokeTransparency = textStrokeTrans
			entry.Label.TextSize = textSize
		end
	end
end

local SeenDebugIds = {}
local function MarkSeen(inst)
	if not inst then return false end
	local ok, id = pcall(function() return inst:GetDebugId() end)
	local key = ok and tostring(id) or tostring(inst)
	if SeenDebugIds[key] then
		return false
	end
	SeenDebugIds[key] = true
	task.delay(6, function()
		SeenDebugIds[key] = nil
	end)
	return true
end

local function ClassifyAndTrackWorldObject(desc)
	if not desc or not desc.Name then return end
	local nameStr = tostring(desc.Name)

	if nameStr == "Door" and desc:IsA("Model") and IsInsideRoom(desc) then
		local room = desc.Parent
		local roomNum = room and tonumber(room.Name)
		local doorText = roomNum and ("Door " .. tostring(roomNum + 1)) or "Door"
		RegisterTrackedObject("Doors", desc, doorText)
		if Toggles.DoorESPToggle and Toggles.DoorESPToggle.Value then
			RefreshCategoryESP("Doors")
		end
	end

	if StructureNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("Structures", desc, StructureNames[nameStr])
		if Toggles.StructureESPToggle and Toggles.StructureESPToggle.Value then
			RefreshCategoryESP("Structures")
		end
	end

	if ObjectiveNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("Objectives", desc, ObjectiveNames[nameStr])
		if Toggles.ObjectiveESPToggle and Toggles.ObjectiveESPToggle.Value then
			RefreshCategoryESP("Objectives")
		end
	end

	if HidingSpotNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("HidingSpots", desc, HidingSpotNames[nameStr])
		if Toggles.HidingSpotESPToggle and Toggles.HidingSpotESPToggle.Value then
			RefreshCategoryESP("HidingSpots")
		end
	end

	if ChestNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("Chests", desc, ChestNames[nameStr])
		if Toggles.ChestESPToggle and Toggles.ChestESPToggle.Value then
			RefreshCategoryESP("Chests")
		end
	end

	if CurrencyNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("Currency", desc, CurrencyNames[nameStr])
		if Toggles.CurrencyESPToggle and Toggles.CurrencyESPToggle.Value then
			RefreshCategoryESP("Currency")
		end
	end

	if LadderNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("Ladders", desc, LadderNames[nameStr])
		if Toggles.LadderESPToggle and Toggles.LadderESPToggle.Value then
			RefreshCategoryESP("Ladders")
		end
	end

	if MiscNames[nameStr] and IsInsideRoom(desc) then
		RegisterTrackedObject("Misc", desc, MiscNames[nameStr])
		if Toggles.MiscESPToggle and Toggles.MiscESPToggle.Value then
			RefreshCategoryESP("Misc")
		end
	end
end

local function OnWorldDescendantAdded(desc)
	if not desc or not desc.Name then return end
	local nameStr = tostring(desc.Name)

	ClassifyAndTrackWorldObject(desc)

	local matchedEntityKey = Entities[nameStr] and nameStr or nil
	if not matchedEntityKey then
		local lowerName = string.lower(nameStr)
		for k in pairs(Entities) do
			if string.find(lowerName, string.lower(k), 1, true) then
				matchedEntityKey = k
				break
			end
		end
	end

	if matchedEntityKey and IsInsideRoom(desc) then
		local targetModel = desc:IsA("Model") and desc or desc:FindFirstAncestorOfClass("Model") or desc
		local alias = ResolveAliasForCurrentMode(matchedEntityKey)
		RegisterTrackedObject("Entities", targetModel, alias)
		if Toggles.EntityESPToggle and Toggles.EntityESPToggle.Value then
			RefreshCategoryESP("Entities")
		end
		if MarkSeen(desc) then
			TriggerEntityNotify(matchedEntityKey)
		end
		return
	end

	if ItemNameSet[nameStr] and IsInsideRoom(desc) and not IsHeldByLocalPlayer(desc) then
		local targetItem = desc:IsA("Model") and desc or desc:FindFirstAncestorOfClass("Model") or desc
		RegisterTrackedObject("Items", targetItem, nameStr)
		if Toggles.ItemESPToggle and Toggles.ItemESPToggle.Value then
			RefreshCategoryESP("Items")
		end
		if MarkSeen(desc) then
			TriggerItemNotify(nameStr)
		end
	end
end

local function HookPlayerCharacter(plr)
	if not plr or plr == LocalPlayer then return end
	local function onChar(char)
		if not char then return end
		RegisterTrackedObject("Players", char, plr.DisplayName or plr.Name)
		if Toggles.PlayerESPToggle and Toggles.PlayerESPToggle.Value then
			RefreshCategoryESP("Players")
		end
	end
	if plr.Character then
		onChar(plr.Character)
	end
	plr.CharacterAdded:Connect(onChar)
end

if Services.Players then
	for _, plr in ipairs(Services.Players:GetPlayers()) do
		HookPlayerCharacter(plr)
	end
	Services.Players.PlayerAdded:Connect(HookPlayerCharacter)
	Services.Players.PlayerRemoving:Connect(function(plr)
		if plr.Character then
			RemoveESP(plr.Character)
			TrackedByCategory.Players[plr.Character] = nil
		end
	end)
end

if workspace then
	workspace.DescendantAdded:Connect(OnWorldDescendantAdded)
	for _, obj in ipairs(workspace:GetDescendants()) do
		OnWorldDescendantAdded(obj)
	end
end

task.spawn(function()
	while task.wait(0.4) do
		if not workspace then continue end
		local currentRooms = workspace:FindFirstChild("CurrentRooms")
		if not currentRooms then continue end
		for _, itemName in ipairs(OrderedItemList) do
			if IsItemSelected(itemName) then
				for _, found in ipairs(currentRooms:GetDescendants()) do
					if found.Name == itemName and not IsHeldByLocalPlayer(found) and MarkSeen(found) then
						TriggerItemNotify(itemName)
					end
				end
			end
		end
	end
end)
