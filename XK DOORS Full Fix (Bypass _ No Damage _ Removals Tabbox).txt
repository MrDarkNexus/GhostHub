local function SafeLoad(Url, Label)
	local tag = tostring(Label or Url)
	if type(Url) ~= "string" or #Url == 0 then
		warn("[SafeLoad] Geçersiz URL (" .. tag .. ")")
		return nil
	end
	local okFetch, sourceOrErr = pcall(function()
		return game:HttpGet(Url)
	end)
	if not okFetch or type(sourceOrErr) ~= "string" or #sourceOrErr == 0 then
		warn("[SafeLoad] İndirme başarısız (" .. tag .. "): " .. tostring(sourceOrErr))
		return nil
	end
	local fn, compileErr = loadstring(sourceOrErr, "@" .. tag)
	if not fn then
		warn("[SafeLoad] Derleme hatası (" .. tag .. "): " .. tostring(compileErr))
		return nil
	end
	local okRun, result = pcall(fn)
	if not okRun then
		warn("[SafeLoad] Çalıştırma hatası (" .. tag .. "): " .. tostring(result))
		return nil
	end
	return result
end

local function getAdornee(obj)
	if typeof(obj) ~= "Instance" then return nil end
	if obj:IsA("BasePart") then return obj end
	if obj:IsA("Model") then
		return obj.PrimaryPart
			or obj:FindFirstChild("HumanoidRootPart")
			or obj:FindFirstChild("Torso")
			or obj:FindFirstChild("UpperTorso")
			or obj:FindFirstChild("Head")
			or obj:FindFirstChild("Core")
			or obj:FindFirstChild("Door")
			or obj:FindFirstChild("Hitbox")
			or obj:FindFirstChildWhichIsA("BasePart", true)
	end
	return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function getEyeCenterPart(obj)
	if typeof(obj) ~= "Instance" then return nil end
	if obj:IsA("BasePart") then return obj end
	if obj:IsA("Model") then
		return obj:FindFirstChild("Core", true)
			or obj:FindFirstChild("Eye", true)
			or obj:FindFirstChild("Eyes", true)
			or obj:FindFirstChild("Head", true)
			or obj.PrimaryPart
			or obj:FindFirstChildWhichIsA("BasePart", true)
	end
	return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function getModelBounds(obj)
	if typeof(obj) ~= "Instance" then return nil, nil end
	if obj:IsA("BasePart") then
		return obj.CFrame, obj.Size
	end
	if obj:IsA("Model") then
		local ok, cf, size = pcall(function()
			return obj:GetBoundingBox()
		end)
		if ok and cf and size and size.Magnitude > 0.05 then
			return cf, size
		end
		local part = getAdornee(obj)
		if part then return part.CFrame, part.Size end
	end
	local part = getAdornee(obj)
	if part then return part.CFrame, part.Size end
	return nil, nil
end

local function detectAutoVisualStyle(obj, opts)
	if opts and opts.Style and tostring(opts.Style) ~= "" then
		return tostring(opts.Style)
	end
	local rawName = typeof(obj) == "Instance" and obj.Name or ""
	local textName = (opts and tostring(opts.Text or "")) or ""
	if DupeVacuumSet[rawName] or DupeVacuumSet[textName] then
		return "DupeVacuumRed"
	end
	if EyesScreechEntitySet[rawName] or EyesScreechEntitySet[textName] then
		return "EyesScreechRing"
	end
	if SmallSquareBoxEntitySet[rawName] or SmallSquareBoxEntitySet[textName] then
		return "SmallSquareBox"
	end
	if SquareBoxEntitySet[rawName] or SquareBoxEntitySet[textName] then
		return "SquareBox"
	end
	if TrapOrHazardShapeSet[rawName] or TrapOrHazardShapeSet[textName] or RedOutlineShapeEntitySet[rawName] or RedOutlineShapeEntitySet[textName] then
		return "RedOutlineShape"
	end
	return "Highlight"
end

	if not entry then return end
	if entry.Highlight then pcall(function() entry.Highlight:Destroy() end) entry.Highlight = nil end
	if entry.Billboard then pcall(function() entry.Billboard:Destroy() end) entry.Billboard = nil end
	if entry.BoxAdornment then pcall(function() entry.BoxAdornment:Destroy() end) entry.BoxAdornment = nil end
	if entry.SquareBoxGui then pcall(function() entry.SquareBoxGui:Destroy() end) entry.SquareBoxGui = nil end
	if entry.CircleBillboard then pcall(function() entry.CircleBillboard:Destroy() end) entry.CircleBillboard = nil end
	if entry.StripeParts then
		for _, p in ipairs(entry.StripeParts) do
			pcall(function() p:Destroy() end)
		end
		entry.StripeParts = nil
	end
end

local function buildWhiteOutlineParts(_, cf, bSize)
	local parts = {}
	if not cf or not bSize then return parts end
	local sx = math.max(bSize.X + 0.16, 0.35)
	local sy = math.max(bSize.Y + 0.16, 0.35)
	local sz = math.max(bSize.Z + 0.16, 0.35)
	local edgeThick = 0.065

	local cornerOffsets = {
		Vector3.new(-sx * 0.5, 0, -sz * 0.5),
		Vector3.new( sx * 0.5, 0, -sz * 0.5),
		Vector3.new(-sx * 0.5, 0,  sz * 0.5),
		Vector3.new( sx * 0.5, 0,  sz * 0.5),
	}
	for _, cOff in ipairs(cornerOffsets) do
		local vEdge = Instance.new("Part")
		vEdge.Name = "XK_WhiteCorner_Edge"
		vEdge.Anchored = true
		vEdge.CanCollide = false
		vEdge.CanQuery = false
		vEdge.CanTouch = false
		vEdge.Material = Enum.Material.Neon
		vEdge.Color = Color3.fromRGB(255, 255, 255)
		vEdge.Transparency = 0
		vEdge.Size = Vector3.new(edgeThick, sy, edgeThick)
		vEdge.CFrame = cf * CFrame.new(cOff)
		vEdge.Parent = Workspace
		table.insert(parts, vEdge)
	end

	for _, ySign in ipairs({ -1, 1 }) do
		for _, zSign in ipairs({ -1, 1 }) do
			local xBar = Instance.new("Part")
			xBar.Name = "XK_WhiteEdge_X"
			xBar.Anchored = true
			xBar.CanCollide = false
			xBar.CanQuery = false
			xBar.CanTouch = false
			xBar.Material = Enum.Material.Neon
			xBar.Color = Color3.fromRGB(255, 255, 255)
			xBar.Transparency = 0
			xBar.Size = Vector3.new(sx, edgeThick, edgeThick)
			xBar.CFrame = cf * CFrame.new(0, ySign * sy * 0.5, zSign * sz * 0.5)
			xBar.Parent = Workspace
			table.insert(parts, xBar)
		end
		for _, xSign in ipairs({ -1, 1 }) do
			local zBar = Instance.new("Part")
			zBar.Name = "XK_WhiteEdge_Z"
			zBar.Anchored = true
			zBar.CanCollide = false
			zBar.CanQuery = false
			zBar.CanTouch = false
			zBar.Material = Enum.Material.Neon
			zBar.Color = Color3.fromRGB(255, 255, 255)
			zBar.Transparency = 0
			zBar.Size = Vector3.new(edgeThick, edgeThick, sz)
			zBar.CFrame = cf * CFrame.new(xSign * sx * 0.5, ySign * sy * 0.5, 0)
			zBar.Parent = Workspace
			table.insert(parts, zBar)
		end
	end
	return parts
end

local function updateWhiteOutlineParts(parts, cf, bSize)
	if not parts or #parts == 0 or not cf or not bSize then return end
	local sx = math.max(bSize.X + 0.16, 0.35)
	local sy = math.max(bSize.Y + 0.16, 0.35)
	local sz = math.max(bSize.Z + 0.16, 0.35)
	local edgeThick = 0.065
	local idx = 1

	for _, cOff in ipairs({
		Vector3.new(-sx * 0.5, 0, -sz * 0.5),
		Vector3.new( sx * 0.5, 0, -sz * 0.5),
		Vector3.new(-sx * 0.5, 0,  sz * 0.5),
		Vector3.new( sx * 0.5, 0,  sz * 0.5),
	}) do
		local p = parts[idx]
		if p and p.Parent then
			p.Size = Vector3.new(edgeThick, sy, edgeThick)
			p.CFrame = cf * CFrame.new(cOff)
			p.Color = Color3.fromRGB(255, 255, 255)
		end
		idx += 1
	end

	for _, ySign in ipairs({ -1, 1 }) do
		for _, zSign in ipairs({ -1, 1 }) do
			local p = parts[idx]
			if p and p.Parent then
				p.Size = Vector3.new(sx, edgeThick, edgeThick)
				p.CFrame = cf * CFrame.new(0, ySign * sy * 0.5, zSign * sz * 0.5)
				p.Color = Color3.fromRGB(255, 255, 255)
			end
			idx += 1
		end
		for _, xSign in ipairs({ -1, 1 }) do
			local p = parts[idx]
			if p and p.Parent then
				p.Size = Vector3.new(edgeThick, edgeThick, sz)
				p.CFrame = cf * CFrame.new(xSign * sx * 0.5, ySign * sy * 0.5, 0)
				p.Color = Color3.fromRGB(255, 255, 255)
			end
			idx += 1
		end
	end
end

function CustomVisualEngine:UpdateObjectColor(obj, color)
	if not obj or not color then return end
	self.ColorTable[obj] = color
	local entry = self.Objects[obj]
	if entry then
		local isAlwaysRed = (entry.Style == "SquareBox" or entry.Style == "SmallSquareBox" or entry.Style == "RedOutlineShape" or entry.Style == "DupeVacuumRed" or entry.Style == "EyesScreechRing")
		local finalCol = isAlwaysRed and Color3.fromRGB(255, 0, 0)
			or (entry.Style == "WhiteOutline" and Color3.fromRGB(255, 255, 255))
			or color
		entry.Options.Color = finalCol
		if entry.Highlight then
			entry.Highlight.FillColor = finalCol
			entry.Highlight.OutlineColor = finalCol
		end
		if entry.BoxAdornment then
			entry.BoxAdornment.Color3 = Color3.fromRGB(255, 0, 0)
		end
		if entry.SquareBoxGui then
			local sqFrame = entry.SquareBoxGui:FindFirstChild("SquareBoxOutline")
			if sqFrame then
				sqFrame.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
				local sqStroke = sqFrame:FindFirstChildOfClass("UIStroke")
				if sqStroke then sqStroke.Color = Color3.fromRGB(255, 0, 0) end
			end
		end
		if entry.CircleBillboard then
			local ring = entry.CircleBillboard:FindFirstChild("EyeCenterRing")
			if ring then
				ring.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
				local stroke = ring:FindFirstChildOfClass("UIStroke")
				if stroke then stroke.Color = Color3.fromRGB(255, 0, 0) end
			end
		end
		if entry.Label then

function CustomVisualEngine:RemoveESP(obj)
	if not obj then return end
	local entry = self.Objects[obj]
	if entry then
		clearCustomVisuals(entry)
		self.Objects[obj] = nil
	end
	self.ColorTable[obj] = nil
end

RunService.Heartbeat:Connect(function()
	local char = LocalPlayer and LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local hue = (tick() * 0.25) % 1
	local rainbowCol = Color3.fromHSV(hue, 1, 1)

	for obj, entry in pairs(CustomVisualEngine.Objects) do
		if not obj or not obj.Parent then
			CustomVisualEngine:RemoveESP(obj)
		else
			if not entry.Adornee or not entry.Adornee.Parent then
				entry.Adornee = entry.Style == "EyesScreechRing"
					and (getEyeCenterPart(obj) or getAdornee(obj))
					or getAdornee(obj)
				if entry.Billboard and entry.Adornee then
					entry.Billboard.Adornee = entry.Adornee
				end
				if entry.CircleBillboard and entry.Adornee then
					entry.CircleBillboard.Adornee = entry.Adornee
				end
				if entry.SquareBoxGui and entry.Adornee then
					entry.SquareBoxGui.Adornee = entry.Adornee
				end
				if entry.BoxAdornment and entry.Adornee then
					entry.BoxAdornment.Adornee = entry.Adornee
				end
			end

			local isAlwaysRed = (entry.Style == "SquareBox" or entry.Style == "SmallSquareBox" or entry.Style == "RedOutlineShape" or entry.Style == "DupeVacuumRed" or entry.Style == "EyesScreechRing")
			local col = isAlwaysRed and Color3.fromRGB(255, 0, 0)
				or (entry.Style == "WhiteOutline" and Color3.fromRGB(255, 255, 255))
				or (CustomVisualEngine.Rainbow and rainbowCol or (entry.Options.Color or Color3.fromRGB(255, 255, 255)))

			if entry.Highlight then
				entry.Highlight.FillColor = col
				entry.Highlight.OutlineColor = col
				if entry.Style == "ItemFullPurple" then
					entry.Highlight.FillTransparency = 0.18
				elseif entry.Style == "DupeVacuumRed" then
					entry.Highlight.FillTransparency = 0.5
				else
					entry.Highlight.FillTransparency = CustomVisualEngine.FillTransparency
				end
				entry.Highlight.OutlineTransparency = CustomVisualEngine.OutlineTransparency
			end

			if entry.StripeParts and #entry.StripeParts > 0 and entry.Style == "WhiteOutline" then
				local cf, bSize = getModelBounds(obj)
				if cf and bSize then
					updateWhiteOutlineParts(entry.StripeParts, cf, bSize)
				end
			end

			if entry.Label then
				entry.Label.TextColor3 = col
				entry.Label.TextTransparency = CustomVisualEngine.TextTransparency
				entry.Label.TextStrokeTransparency = CustomVisualEngine.TextOutlineTransparency
				entry.Label.TextSize = CustomVisualEngine.TextSize
				entry.Label.Font = CustomVisualEngine.Font

				local distStr = ""
				if CustomVisualEngine.ShowDistance and hrp and entry.Adornee and entry.Adornee.Parent then
					local d = math.floor((hrp.Position - entry.Adornee.Position).Magnitude)
					distStr = " [" .. d .. "]"
				end
				entry.Label.Text = tostring(entry.Options.Text or entry.BaseText) .. distStr
			end
		end
	end
end)

	if not Translations[k] then
		Translations[k] = { tr = v.tr, en = v.en }

local CurrentLang = "tr"
local TranslatableElements = {}
local RefreshAllESPLabels = nil

local function GetESPName(rawKey, lang)
	local activeLang = lang or CurrentLang or "tr"
	local entry = ESPNameTranslations[rawKey]
	if entry and entry[activeLang] then
		return entry[activeLang]
	end
	return tostring(rawKey)
end

local function GetTranslatedText(originalText, lang)
	if lang == "zh" or type(originalText) ~= "string" then
		return originalText
	end
	local entry = Translations[originalText]
	if entry and entry[lang] then
		return entry[lang]
	end
	return originalText
end

local function RegisterTranslatable(setFn, getFn)
	table.insert(TranslatableElements, { set = setFn, get = getFn })
end

local function ApplyLanguage(lang)
	CurrentLang = (lang == "tr" or lang == "en" or lang == "zh") and lang or "tr"

	for _, item in ipairs(TranslatableElements) do
		pcall(function()
			local orig = item.get()
			if type(orig) == "string" then
				item.set(GetTranslatedText(orig, CurrentLang))
			end
		end)
	end

	local root = UILibrary.ScreenGui or CoreGui
	if root then
		for _, desc in ipairs(root:GetDescendants()) do
			if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
				if not desc:GetAttribute("OrigText") then
					desc:SetAttribute("OrigText", desc.Text)
				end
				local origText = desc:GetAttribute("OrigText")
				local mapped = GetTranslatedText(origText, CurrentLang)
				if mapped ~= origText then
					desc.Text = mapped
				elseif CurrentLang == "zh" then
					desc.Text = origText
				end
			end
		end
	end

	if type(RefreshAllESPLabels) == "function" then
		pcall(RefreshAllESPLabels)
	end
end

local function WrapTextContainer(container, origText)
	if not container then return end
	if type(container.SetText) == "function" then
		local current = origText
		RegisterTranslatable(
			function(txt) container:SetText(txt) end,
			function() return current end
		)
	end
end

local function GetCurrentFloor()
	local fl = ""
	pcall(function()
		local gd = ReplicatedStorage:FindFirstChild("GameData")
		local fObj = gd and gd:FindFirstChild("Floor")
		if fObj then fl = tostring(fObj.Value) end

local function ApplyAccelerationState(char, enabled)
	if not char then return end
	for _, part in ipairs(char:GetDescendants()) do
		if part:IsA("BasePart") then
			if not PartOriginalProperties[part] then
				PartOriginalProperties[part] = part.CustomPhysicalProperties
			end
			if enabled then
				part.CustomPhysicalProperties = NoAccelPhysicalProps
			else
				part.CustomPhysicalProperties = PartOriginalProperties[part] or DefaultPhysicalProps
			end
		end
	end
end

local function ApplyJumpAndSlideState(char, hum)
	if char then
		char:SetAttribute("CanJump", EnableJumpEnabled and true or false)
		char:SetAttribute("CanSlide", EnableSlidingEnabled and true or false)
	end
	if hum and EnableJumpEnabled then
		pcall(function()
			hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
			if hum.JumpHeight < 7.2 then
				hum.JumpHeight = 7.2
			end
		end)
	end
end

local function IsPositionCrouchDangerActive()
	for _, child in ipairs(Workspace:GetChildren()) do
		if PositionCrouchEntities[child.Name] then
			return true
		end
	end
	return false
end

local function ForceCrouchPositionNow(char, hum, hrp)
	if not char or not hum or not hrp then return end
	if OriginalHipHeight == nil then
		OriginalHipHeight = hum.HipHeight
	end
	local lowerTorso = char:FindFirstChild("LowerTorso")
	local rootJoint = lowerTorso and lowerTorso:FindFirstChild("Root")
	if rootJoint and OriginalRootC1 == nil then
		OriginalRootC1 = rootJoint.C1
	end

	pcall(function()
		char:SetAttribute("Crouching", true)
		hum.HipHeight = 0.15
		if rootJoint and OriginalRootC1 then
			rootJoint.C1 = OriginalRootC1 * CFrame.new(0, -1.85, 0)
		end
		if IsPositionCrouchDangerActive() then
			for _, p in ipairs(char:GetDescendants()) do
				if p:IsA("BasePart") then
					p.CanCollide = false
				end
			end
			char:PivotTo(hrp.CFrame * CFrame.new(0, -5.5, 0))
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
	end)
end

local function ApplyPositionCrouch(enabled)
	PositionCrouchEnabled = enabled
	local char = LocalPlayer and LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not char or not hum or not hrp then return end

	if OriginalHipHeight == nil then
		OriginalHipHeight = hum.HipHeight
	end
	local lowerTorso = char:FindFirstChild("LowerTorso")
	local rootJoint = lowerTorso and lowerTorso:FindFirstChild("Root")
	if rootJoint and OriginalRootC1 == nil then
		OriginalRootC1 = rootJoint.C1
	end

	pcall(function()
		if enabled then
			ForceCrouchPositionNow(char, hum, hrp)
			local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
			if remotes and remotes:FindFirstChild("Crouch") then
				remotes.Crouch:FireServer(true)
			end
		else
			char:SetAttribute("Crouching", false)
			hum.HipHeight = OriginalHipHeight or 2.396
			if rootJoint and OriginalRootC1 then
				rootJoint.C1 = OriginalRootC1
			end
			local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
			if remotes and remotes:FindFirstChild("Crouch") then
				remotes.Crouch:FireServer(false)
			end
		end
	end)
end

	local char = LocalPlayer and LocalPlayer.Character

		end
	end)
end

local function SetupCharacterRig(newChar)
	if not newChar then return end
	Character = newChar
	Humanoid = newChar:WaitForChild("Humanoid", 10)
	RootPart = newChar:WaitForChild("HumanoidRootPart", 10)
	CollisionPart = newChar:FindFirstChild("CollisionPart") or newChar:FindFirstChild("Collision") or RootPart
	Camera = Workspace.CurrentCamera

	if Humanoid then
		OriginalHipHeight = Humanoid.HipHeight
	end
	local lowerTorso = newChar:FindFirstChild("LowerTorso")
	local rootJoint = lowerTorso and lowerTorso:FindFirstChild("Root")
	if rootJoint then
		OriginalRootC1 = rootJoint.C1
	end

	if FlyBodyVelocity then pcall(function() FlyBodyVelocity:Destroy() end) end
	FlyBodyVelocity = Instance.new("BodyVelocity")
	FlyBodyVelocity.Name = "XK_FlyVelocity"
	FlyBodyVelocity.MaxForce = Vector3.new(1e7, 1e7, 1e7)
	FlyBodyVelocity.Velocity = Vector3.zero

	if VelocityManipulationBody then pcall(function() VelocityManipulationBody:Destroy() end) end
	VelocityManipulationBody = Instance.new("BodyVelocity")
	VelocityManipulationBody.Name = "XK_VelocityManipulation"
	VelocityManipulationBody.MaxForce = Vector3.new(1e7, 0, 1e7)
	VelocityManipulationBody.Velocity = Vector3.zero

	ApplyAccelerationState(newChar, RemoveAccelerationEnabled)
	ApplyJumpAndSlideState(newChar, Humanoid)
	if PositionCrouchEnabled then
		ApplyPositionCrouch(true)
	end
	if SpoofitionCrouchEnabled then
		ApplySpoofitionCrouch(true)
	end
end

if LocalPlayer then
	if LocalPlayer.Character then
		task.spawn(SetupCharacterRig, LocalPlayer.Character)
	end
	LocalPlayer.CharacterAdded:Connect(SetupCharacterRig)
end

local function IsAllowedAutoInteractPrompt(prompt)
	if not prompt or not prompt:IsA("ProximityPrompt") then return false end
	local pName = string.lower(prompt.Name)
	local pAction = string.lower(prompt.ActionText or "")

	if string.find(pName, "hide", 1, true)
		or string.find(pName, "sit", 1, true)
		or string.find(pName, "bed", 1, true)
		or string.find(pAction, "hide", 1, true)
		or string.find(pAction, "saklan", 1, true)
		or string.find(pAction, "sit", 1, true)
		or string.find(pAction, "otur", 1, true)
		or string.find(pAction, "yatak", 1, true)
	then
		return false
	end

	local cur = prompt.Parent
	local depth = 0
	while cur and cur ~= Workspace and depth < 7 do
		if HidingSpotFilterNames[cur.Name] then
			return false
		end
		if AutoInteractTargetFilterNames[cur.Name] then
			return true
		end
		local cLower = string.lower(cur.Name)
		if string.find(cLower, "drawer", 1, true)
			or string.find(cLower, "cubby", 1, true)
			or string.find(cLower, "chest", 1, true)
			or string.find(cLower, "toolbox", 1, true)
			or string.find(cLower, "locker_small", 1, true)
		then
			return true
		end
		cur = cur.Parent
		depth += 1
	end
	return false
end

local function GetPromptPosition(prompt)
	local parent = prompt and prompt.Parent
	if not parent then return nil end
	if parent:IsA("BasePart") then return parent.Position end
	if parent:IsA("Attachment") then return parent.WorldPosition end
	if parent:IsA("Model") then
		local ok, cf = pcall(function() return parent:GetPivot() end)
		return ok and cf and cf.Position or nil
	end
	local part = parent:FindFirstChildWhichIsA("BasePart", true)
	return part and part.Position or nil
end

local function BoostPrompt(prompt)
	if not prompt or not prompt:IsA("ProximityPrompt") then return end
	if not prompt:GetAttribute("OrigHoldDuration") then
		prompt:SetAttribute("OrigHoldDuration", prompt.HoldDuration)
	end
	if not prompt:GetAttribute("OrigMaxDist") then
		prompt:SetAttribute("OrigMaxDist", prompt.MaxActivationDistance)
	end
	if not prompt:GetAttribute("OrigLineOfSight") then
		prompt:SetAttribute("OrigLineOfSight", prompt.RequiresLineOfSight)
	end

	if PromptRageEnabled then
		prompt.HoldDuration = 0
		prompt.RequiresLineOfSight = false
		local baseDist = prompt:GetAttribute("OrigMaxDist") or prompt.MaxActivationDistance
		prompt.MaxActivationDistance = math.max(baseDist * 1.75, 16)
	else
		prompt.HoldDuration = prompt:GetAttribute("OrigHoldDuration") or prompt.HoldDuration
		prompt.MaxActivationDistance = prompt:GetAttribute("OrigMaxDist") or prompt.MaxActivationDistance
		prompt.RequiresLineOfSight = (prompt:GetAttribute("OrigLineOfSight") ~= nil) and prompt:GetAttribute("OrigLineOfSight") or prompt.RequiresLineOfSight
	end
end

local function FindNextClosedDoor()
	local gd = ReplicatedStorage:FindFirstChild("GameData")
	local latest = gd and gd:FindFirstChild("LatestRoom")
	local curRooms = Workspace:FindFirstChild("CurrentRooms")
	if not latest or not curRooms then return nil, nil end

	local startRoom = tonumber(latest.Value) or 0
	local bestDoor, bestNum = nil, math.huge
	for _, room in ipairs(curRooms:GetChildren()) do
		local num = tonumber(room.Name)
		if num and num >= startRoom and num < bestNum then
			local door = room:FindFirstChild("Door")
			if door and door:IsA("Model") and door:GetAttribute("Opened") ~= true then
				bestNum = num
				bestDoor = door
			end
		end
	end
	return bestDoor, bestNum
end

local lastAutoInteractTick = 0
RunService.Heartbeat:Connect(function(dt)
	local char = LocalPlayer and LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local cam = Workspace.CurrentCamera

	if WalkSpeedEnabled and hum then
		hum.WalkSpeed = WalkSpeedTarget
	end

	if PositionCrouchEnabled and char and hum and hrp then
		ForceCrouchPositionNow(char, hum, hrp)
	end

	if Options and Options.VelocityModeSelect and Options.VelocityModeSelect.Value then
		VelocityModeValue = tostring(Options.VelocityModeSelect.Value)
	end

	if VelocityManipulationEnabled and char and hrp and hum then
		local moveDir = hum.MoveDirection
		local spd = (WalkSpeedEnabled and WalkSpeedTarget) or hum.WalkSpeed
		if not spd or spd <= 0 then spd = 16 end

		if VelocityModeValue == "Velocity" then
			if not VelocityManipulationBody or not VelocityManipulationBody.Parent then
				pcall(function()
					if VelocityManipulationBody then VelocityManipulationBody:Destroy() end
				end)
				VelocityManipulationBody = Instance.new("BodyVelocity")
				VelocityManipulationBody.Name = "XK_VelocityManipulation"
				VelocityManipulationBody.MaxForce = Vector3.new(1e7, 0, 1e7)
				VelocityManipulationBody.P = 1250
			end
			VelocityManipulationBody.Parent = hrp
			local targetVel = Vector3.new(moveDir.X * spd, 0, moveDir.Z * spd)
			VelocityManipulationBody.Velocity = targetVel
			if moveDir.Magnitude > 0.01 then
				pcall(function()
					hrp.AssemblyLinearVelocity = Vector3.new(targetVel.X, hrp.AssemblyLinearVelocity.Y, targetVel.Z)
				end)
			end
		elseif VelocityModeValue == "Pivot" then
			if VelocityManipulationBody and VelocityManipulationBody.Parent then
				VelocityManipulationBody.Parent = nil
				VelocityManipulationBody.Velocity = Vector3.zero
			end
			if moveDir.Magnitude > 0.01 then
				pcall(function()
					local deltaPos = moveDir.Unit * spd * math.clamp(dt or 0.016, 0.001, 0.1)
					local currentPivot = char:GetPivot()
					char:PivotTo(currentPivot + deltaPos)
				end)
			end
		end
	elseif VelocityManipulationBody and VelocityManipulationBody.Parent then
		VelocityManipulationBody.Parent = nil
		VelocityManipulationBody.Velocity = Vector3.zero
	end

	if NoClipEnabled and char then
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				part.CanCollide = false
			end
		end
		if CollisionPart and CollisionPart:IsA("BasePart") then
			CollisionPart.CanCollide = false
		end
	end

	if FlyEnabled and hrp and hum and cam then
		if not FlyBodyVelocity or not FlyBodyVelocity.Parent then
			FlyBodyVelocity = Instance.new("BodyVelocity")
			FlyBodyVelocity.Name = "XK_FlyVelocity"
			FlyBodyVelocity.MaxForce = Vector3.new(1e7, 1e7, 1e7)
		end
		FlyBodyVelocity.Parent = hrp

		local moveDir = hum.MoveDirection
		local flyVec = Vector3.zero
		if moveDir.Magnitude > 0.01 then
			local lookFlat = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z)
			if lookFlat.Magnitude > 0.001 then
				local flatFrame = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + lookFlat.Unit)
				local localMove = flatFrame:VectorToObjectSpace(moveDir)
				flyVec = (cam.CFrame:VectorToWorldSpace(localMove))
			else
				flyVec = moveDir
			end
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) and not IsTyping then
			flyVec = flyVec + Vector3.new(0, 1, 0)
		end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) and not IsTyping then
			flyVec = flyVec - Vector3.new(0, 1, 0)
		end
		if flyVec.Magnitude > 0.001 then
			FlyBodyVelocity.Velocity = flyVec.Unit * FlySpeedValue
		else
			FlyBodyVelocity.Velocity = Vector3.zero
		end
	elseif FlyBodyVelocity and FlyBodyVelocity.Parent then
		FlyBodyVelocity.Parent = nil
		FlyBodyVelocity.Velocity = Vector3.zero
	end

	if AutoInteractEnabled and hrp and tick() - lastAutoInteractTick >= 0.08 then
		lastAutoInteractTick = tick()
		local curRooms = Workspace:FindFirstChild("CurrentRooms")
		local searchRoot = curRooms or Workspace
		for _, desc in ipairs(searchRoot:GetDescendants()) do
			if desc:IsA("ProximityPrompt") and desc.Enabled then
				if IsAllowedAutoInteractPrompt(desc) then
					local pPos = GetPromptPosition(desc)
					if pPos and (hrp.Position - pPos).Magnitude <= math.max(desc.MaxActivationDistance, 14) then
						pcall(function()
							if fireproximityprompt then
								fireproximityprompt(desc)
							end
						end)
					end
				end
			end
		end
	end
end)

local function SetRemoteIntercepted(remoteName, enabled)
	local rf = GetActiveRemotesFolder()
	if not rf then return end
	local fake = GetOrCreateFakeRemote(remoteName)
	local real = CachedRealRemotes[remoteName]
	if not real or not real.Parent then
		local found = rf:FindFirstChild(remoteName)
		if found and found ~= fake then
			real = found
			CachedRealRemotes[remoteName] = found
		end
	end
	pcall(function()
		if enabled then
			if real and real.Parent == rf then
				real.Parent = nil
			end
			fake.Parent = rf
		else
			if fake.Parent == rf then
				fake.Parent = nil
			end
			if real and real.Parent ~= rf then
				real.Parent = rf
			end
		end
	end)
end

				SetPartTouchAndCollide(d, false, nil)

	end

		for _, d in ipairs(inst:GetDescendants()) do
			if d:IsA("BasePart") then
				SetPartTouchAndCollide(d, false, nil)
			end
		end
	end

	if BypassStates.Vacuum and (name == "SideroomSpace" or name == "Vacuum" or string.find(string.lower(name), "vacuum", 1, true)) then
		for _, d in ipairs(inst:GetDescendants()) do
			if d:IsA("BasePart") then
				if d.Name == "Collision" then
					SetPartTouchAndCollide(d, false, true)
				else
					SetPartTouchAndCollide(d, false, nil)
				end
			end
		end

			if d:IsA("BasePart") then
				SetPartTouchAndCollide(d, false, nil)
			end
		end
	end

	if BypassStates.KillBricks and (name == "Lava" or string.find(string.lower(name), "killbrick", 1, true) or string.find(string.lower(name), "lava", 1, true) or string.find(string.lower(name), "spike", 1, true)) then

			end
		end
	end

	if BypassStates.Banana and (name == "BananaPeel" or string.find(string.lower(name), "banana", 1, true)) then
		if inst:IsA("BasePart") then
			SetPartTouchAndCollide(inst, false, nil)
		end
		for _, d in ipairs(inst:GetDescendants()) do
			if d:IsA("BasePart") then
				SetPartTouchAndCollide(d, false, nil)
			end
		end
	end

	if BypassStates.JeffTheKiller and (name == "JeffTheKiller" or string.find(string.lower(name), "jeff", 1, true)) then
		for _, d in ipairs(inst:GetDescendants()) do
			if d:IsA("BasePart") then
				SetPartTouchAndCollide(d, false, false)
			elseif d:IsA("Humanoid") then

			end
		end

			or string.find(lower, "seekhand", 1, true)
			or string.find(lower, "seekfire", 1, true)

local function RestoreBypassState(stateKey)
	local rooms = Workspace:FindFirstChild("CurrentRooms")
	local roots = { Workspace }
	if rooms then table.insert(roots, rooms) end

	for _, root in ipairs(roots) do
		for _, desc in ipairs(root:GetDescendants()) do
			local n = desc.Name
			local match = false
			if stateKey == "Giggle" and (n == "GiggleCeiling" or n == "Giggle") then match = true
			elseif stateKey == "Dupe" and (n == "DoorFake" or n == "FakeDoor" or string.find(string.lower(n), "dupe", 1, true)) then match = true
			elseif stateKey == "Snare" and (n == "Snare" or string.find(string.lower(n), "snare", 1, true)) then match = true
			elseif stateKey == "Vacuum" and (n == "SideroomSpace" or n == "Vacuum" or string.find(string.lower(n), "vacuum", 1, true)) then match = true
			elseif stateKey == "GlomBatsEgg" and (n == "GloomPile" or string.find(string.lower(n), "gloom", 1, true) or string.find(string.lower(n), "egg", 1, true)) then match = true
			elseif stateKey == "KillBricks" and (n == "Lava" or string.find(string.lower(n), "killbrick", 1, true) or string.find(string.lower(n), "lava", 1, true)) then match = true
			elseif stateKey == "Banana" and (n == "BananaPeel" or string.find(string.lower(n), "banana", 1, true)) then match = true
			elseif stateKey == "JeffTheKiller" and (n == "JeffTheKiller" or string.find(string.lower(n), "jeff", 1, true)) then match = true
			elseif stateKey == "SeekHandFire" and (n == "SeekObstructions" or n == "SeekBridges" or string.find(string.lower(n), "seek", 1, true) or string.find(string.lower(n), "fire", 1, true)) then match = true
			elseif stateKey == "SeekingWall" and (n == "ScaryWall" or string.find(string.lower(n), "seekingwall", 1, true) or string.find(string.lower(n), "scarywall", 1, true)) then match = true
			end

			if match then
				if desc:IsA("BasePart") then
					SetPartTouchAndCollide(desc, nil, nil)
				elseif desc:IsA("ProximityPrompt") then
					SetPromptEnabledSafe(desc, nil)
				end
				for _, sub in ipairs(desc:GetDescendants()) do
					if sub:IsA("BasePart") then
						SetPartTouchAndCollide(sub, nil, nil)
					elseif sub:IsA("ProximityPrompt") then
						SetPromptEnabledSafe(sub, nil)
					end
				end
			end
		end
	end
end

local function ScanAllBypassTargets()
	for _, child in ipairs(Workspace:GetChildren()) do
		ApplyBypassToInstance(child)
	end
	local rooms = Workspace:FindFirstChild("CurrentRooms")
	if rooms then
		for _, room in ipairs(rooms:GetChildren()) do
			ApplyBypassToInstance(room)
			for _, desc in ipairs(room:GetDescendants()) do
				ApplyBypassToInstance(desc)
			end
		end
	end
end

local function SetEntityRemovedState(entityKey, enabled)
	RemoveStates[entityKey] = enabled

	local moduleNames = {}
	if entityKey == "Surge" then
		moduleNames = { "Surge", "SurgeClient", "SurgeSpawn" }
	elseif entityKey == "A90" then
		moduleNames = { "A90", "A-90" }
	elseif entityKey == "Dread" then
		moduleNames = { "Dread" }
	elseif entityKey == "Halt" then
		moduleNames = { "Shade", "Halt" }
	end

	local searchContainers = {
		ReplicatedStorage:FindFirstChild("ClientModules"),
		ReplicatedStorage:FindFirstChild("Modules"),
		ReplicatedStorage:FindFirstChild("FloorReplicated"),
		Workspace:FindFirstChild("Camera"),
	}

	for _, modName in ipairs(moduleNames) do
		local cacheKey = entityKey .. "_" .. modName
		if enabled then
			for _, container in ipairs(searchContainers) do
				if container then
					for _, obj in ipairs(container:GetDescendants()) do
						if obj.Name == modName or obj.Name == (modName .. "_Disabled") then
							if not CachedRemovedModules[cacheKey] then
								CachedRemovedModules[cacheKey] = {}
							end
							table.insert(CachedRemovedModules[cacheKey], {
								inst = obj,
								origName = obj.Name,
								origParent = obj.Parent,
							})
							pcall(function()
								obj.Name = modName .. "_Disabled"
								if obj:IsA("GuiObject") then
									obj.Visible = false
								end
							end)
						end
					end
				end
			end
			for _, wObj in ipairs(Workspace:GetDescendants()) do
				if wObj.Name == modName then
					pcall(function() wObj:Destroy() end)
				end
			end
			local pGui = LocalPlayer and LocalPlayer:FindFirstChild("PlayerGui")
			if pGui then
				for _, gObj in ipairs(pGui:GetDescendants()) do
					if gObj.Name == modName or string.find(string.lower(gObj.Name), string.lower(modName), 1, true) then
						if gObj:IsA("GuiObject") then
							if not CachedRemovedFrames[gObj] then
								CachedRemovedFrames[gObj] = gObj.Visible
							end
							pcall(function() gObj.Visible = false end)
						end
					end
				end
			end
		else
			local list = CachedRemovedModules[cacheKey]
			if list then
				for _, entry in ipairs(list) do
					if entry.inst and entry.origName then
						pcall(function()
							entry.inst.Name = entry.origName
							if entry.origParent and entry.origParent.Parent and entry.inst.Parent ~= entry.origParent then
								entry.inst.Parent = entry.origParent
							end
						end)
					end
				end
				CachedRemovedModules[cacheKey] = nil
			end
		end
	end
end

local function IsCameraFirstPerson(cam)
	if not cam then return false end
	return (cam.CFrame.Position - cam.Focus.Position).Magnitude <= 1.5
end

local function NeutralizeCameraShake(cam, char)
	if not NoCameraShakeEnabled then return end
	pcall(function()
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.CameraOffset = Vector3.zero
		end
		if cam then
			local shakeVal = cam:FindFirstChild("CameraShake") or cam:FindFirstChild("Shake")
			if shakeVal and (shakeVal:IsA("NumberValue") or shakeVal:IsA("Vector3Value")) then
				if shakeVal:IsA("NumberValue") then
					shakeVal.Value = 0
				elseif shakeVal:IsA("Vector3Value") then
					shakeVal.Value = Vector3.zero
				end
			end
		end
	end)
end

		return true
	end

	end

local function IsElevatorOrBoardedDoor(obj)
	if not obj then return false end
	local nameLower = string.lower(tostring(obj.Name or ""))
	if string.find(nameLower, "elevator", 1, true)
		or string.find(nameLower, "board", 1, true)
		or string.find(nameLower, "plank", 1, true)
		or string.find(nameLower, "barricade", 1, true)
	then
		return true
	end

	local cur = obj
	local depth = 0
	while cur and cur ~= Workspace and depth < 6 do
		local cLower = string.lower(tostring(cur.Name or ""))
		if string.find(cLower, "elevator", 1, true)
			or string.find(cLower, "board", 1, true)
			or string.find(cLower, "plank", 1, true)
			or string.find(cLower, "barricade", 1, true)
		then
			return true
		end
		cur = cur.Parent
		depth += 1
	end

	for _, desc in ipairs(obj:GetDescendants()) do
		local dLower = string.lower(tostring(desc.Name or ""))
		if string.find(dLower, "plank", 1, true)
			or string.find(dLower, "board", 1, true)
			or string.find(dLower, "barricade", 1, true)
			or string.find(dLower, "elevator", 1, true)
		then
			return true
		end
	end

	return false
end

local function ResolveDoorLabel(doorModel)
	local floor = GetFloorValue()
	local num = tonumber(doorModel.Parent and doorModel.Parent.Name)
		or tonumber(doorModel.Parent and doorModel.Parent.Parent and doorModel.Parent.Parent.Name)

	if num then
		num = num + 1
		if floor == "Mines" then
			num = num + 100
		elseif floor == "Backdoor" then
			num = num - 50
		end
	end

	local doorPrefix = (CurrentLang == "tr") and "Kapı " or "Door "
	local baseTitle = num and (doorPrefix .. tostring(num)) or GetESPName("门")

	local isOpened = (doorModel:GetAttribute("Opened") == true) or (doorModel:GetAttribute("Open") == true)
	local isLocked = (doorModel:GetAttribute("Locked") == true) or (doorModel:FindFirstChild("Lock", true) ~= nil)

	local statusTag = nil
	if isOpened then
		statusTag = "[Opened]"
	elseif isLocked then

	if statusTag then
		return baseTitle .. "\n" .. statusTag
	end
	return baseTitle
end

local function ClearObjectESP(obj)
	if not obj then return end
	if ESPController.RoomConnections[obj] then
		for _, conn in ipairs(ESPController.RoomConnections[obj]) do
			pcall(function() conn:Disconnect() end)
		end
		ESPController.RoomConnections[obj] = nil
	end
	ESPController.ActiveESP[obj] = nil
	pcall(function()
		Abysall.ESPLibrary:RemoveESP(obj)
	end)
end

local function ApplyObjectESP(obj, rawKeyOrText, color, roomBased, dynamicTextFn, styleOverride)
	if not obj or not obj.Parent then return end
	ClearObjectESP(obj)

	local style = styleOverride or obj:GetAttribute("ESPStyle") or "Highlight"
	local keepWhenDropped = (obj:GetAttribute("KeepWhenDropped") == true) or (style == "ItemFullPurple")

	local function buildText()
		if type(dynamicTextFn) == "function" then
			return dynamicTextFn(obj)
		end
		local rawKey = obj:GetAttribute("ESPRawKey") or rawKeyOrText or tostring(obj.Name)
		local localized = GetESPName(rawKey)
		local extraSuffix = obj:GetAttribute("ESPExtraSuffix")
		if extraSuffix and tostring(extraSuffix) ~= "" then
			return localized .. " " .. tostring(extraSuffix)
		end
		return localized
	end

	local function refresh()
		if not obj or not obj.Parent then
			ClearObjectESP(obj)
			return
		end
		local label = buildText()
		if not label then
			pcall(function() Abysall.ESPLibrary:RemoveESP(obj) end)
			return
		end
		if keepWhenDropped or IsRoomVisible(obj, roomBased) then
			pcall(function()
				Abysall.ESPLibrary:AddESP({
					Object = obj,
					Text = label,
					Color = color,
					Style = style,
				})
			end)
		else
			pcall(function() Abysall.ESPLibrary:RemoveESP(obj) end)
		end
	end

	ESPController.ActiveESP[obj] = {
		RawKey = rawKeyOrText,
		Color = color,
		RoomBased = roomBased,
		DynamicTextFn = dynamicTextFn,
		Style = style,
		Refresh = refresh,
	}

	refresh()

	local conns = {}
	if roomBased and LocalPlayer and not keepWhenDropped then
		table.insert(conns, LocalPlayer:GetAttributeChangedSignal("CurrentRoom"):Connect(refresh))
	end
	table.insert(conns, obj:GetAttributeChangedSignal("Opened"):Connect(refresh))
	table.insert(conns, obj:GetAttributeChangedSignal("Open"):Connect(refresh))
	table.insert(conns, obj:GetAttributeChangedSignal("Locked"):Connect(refresh))
	table.insert(conns, obj:GetAttributeChangedSignal("ParentRoom"):Connect(refresh))
	table.insert(conns, obj.AncestryChanged:Connect(function()
		if not obj or not obj.Parent then
			ClearObjectESP(obj)
		elseif keepWhenDropped then
			task.defer(refresh)
		end
	end))
	table.insert(conns, obj.DescendantAdded:Connect(function(d)
		if d.Name == "Lock" then
			refresh()
		end
	end))
	table.insert(conns, obj.Destroying:Once(function()
		ClearObjectESP(obj)
	end))
	ESPController.RoomConnections[obj] = conns
end

local VALID_CUBBIES = {
	Cubby_Large = true,
	Cubby_Tall = true,
	Cubby_Small = true,
	Small_Cubby = true,
	Cubby_Medium = true,
}
local CubbyTracked = {}
local CubbyMainConn = nil

local function getCubbyPosition(cubby)
	if typeof(cubby) ~= "Instance" then return nil end
	if cubby:IsA("BasePart") then return cubby.Position end
	if cubby:IsA("Model") then
		if cubby.PrimaryPart then return cubby.PrimaryPart.Position end
		local ok, cf = pcall(function() return cubby:GetPivot() end)
		if ok and cf then return cf.Position end
		local p = cubby:FindFirstChildWhichIsA("BasePart", true)
		if p then return p.Position end
	end
	local p = cubby:FindFirstChildWhichIsA("BasePart", true)
	return p and p.Position or nil
end

local function isAdjacentToAnotherCubby(cubby, allCubbies)
	local posA = getCubbyPosition(cubby)
	if not posA then return false end
	for _, other in ipairs(allCubbies) do
		if other ~= cubby and other and other.Parent then
			local posB = getCubbyPosition(other)
			if posB and (posA - posB).Magnitude <= 8.5 then
				return true
			end
		end
	end
	return false
end

local function ResolveCubbyDisplayName(cubby)
	if not cubby then return "Cubby_Small" end
	local rawName = tostring(cubby.Name or "")
	if rawName == "Small_Cubby" then
		rawName = "Cubby_Small"
	end

	local attrCount = tonumber(cubby:GetAttribute("CubbyCount")) or tonumber(cubby:GetAttribute("StackCount"))
	local hasTwoTag = (attrCount and attrCount >= 2)
		or string.find(string.lower(rawName), "(2)", 1, true) ~= nil
		or cubby:FindFirstChild("Cubby2", true) ~= nil
		or cubby:FindFirstChild("SecondCubby", true) ~= nil

	local adjacentList = {}
	local parentContainer = cubby.Parent
	if parentContainer then
		for _, child in ipairs(parentContainer:GetChildren()) do
			if VALID_CUBBIES[child.Name] then
				table.insert(adjacentList, child)
			end
		end
	end

	local isSideBySide = hasTwoTag or isAdjacentToAnotherCubby(cubby, adjacentList)
	if isSideBySide then
		return "Cubby_Small"
	end

	return rawName
end

local function IsCubbyLooted(cubby)
	if not cubby or not cubby.Parent then return true end
	local lootables = {}
	for _, child in ipairs(cubby:GetChildren()) do
		if child.Name == "LootableCubby" then
			table.insert(lootables, child)
		end
	end
	if #lootables == 0 then
		return false
	end
	local opened = 0
	for _, lootable in ipairs(lootables) do
		if lootable:FindFirstChild("LootHolder", true) then
			opened += 1
		end
	end
	return opened >= #lootables
end

local function RemoveCubbyESP(cubby)
	local data = CubbyTracked[cubby]
	if data then
		for _, c in ipairs(data.conns or {}) do
			pcall(function() c:Disconnect() end)
		end
		CubbyTracked[cubby] = nil
	end
	ClearObjectESP(cubby)
end

local function SyncAllCubbiesInRoom(rootParent)
	if not rootParent then return end
	for _, desc in ipairs(rootParent:GetDescendants()) do
		if VALID_CUBBIES[desc.Name] and CubbyTracked[desc] then
			local resolved = ResolveCubbyDisplayName(desc)
			local col = (Options.CubbyESPColor and Options.CubbyESPColor.Value) or Color3.fromRGB(255, 105, 180)
			pcall(function()
				desc:SetAttribute("ESPRawKey", resolved)
				desc:SetAttribute("ESPStyle", "Highlight")
			end)
			ApplyObjectESP(desc, resolved, col, true, function(targetObj)
				return ResolveCubbyDisplayName(targetObj)
			end, "Highlight")
		end
	end
end

local function AddCubbyESP(cubby)
	if not Toggles.CubbyESPToggle or not Toggles.CubbyESPToggle.Value then return end
	if not VALID_CUBBIES[cubby.Name] then return end
	if IsCubbyLooted(cubby) then
		RemoveCubbyESP(cubby)
		return
	end

	local col = (Options.CubbyESPColor and Options.CubbyESPColor.Value) or Color3.fromRGB(255, 105, 180)
	local resolvedName = ResolveCubbyDisplayName(cubby)

	pcall(function()
		cubby:SetAttribute("ESPRawKey", resolvedName)
		cubby:SetAttribute("ESPStyle", "Highlight")
	end)
	ApplyObjectESP(cubby, resolvedName, col, true, function(targetObj)
		return ResolveCubbyDisplayName(targetObj)
	end, "Highlight")

	if CubbyTracked[cubby] then return end

	local conns = {}
	local function checkState()
		if IsCubbyLooted(cubby) then
			RemoveCubbyESP(cubby)
		else
			task.defer(function()
				if cubby.Parent then
					SyncAllCubbiesInRoom(cubby.Parent.Parent or cubby.Parent)
				end
			end)
		end
	end
	table.insert(conns, cubby.DescendantAdded:Connect(function(d)
		if d.Name == "LootHolder" or d.Name == "LootableCubby" or d.Name == "Cubby2" or d.Name == "SecondCubby" then
			task.defer(checkState)
		end
	end))
	table.insert(conns, cubby.DescendantRemoving:Connect(function(d)
		if d.Name == "LootHolder" or d.Name == "LootableCubby" then
			task.defer(checkState)
		end
	end))
	table.insert(conns, cubby.Destroying:Once(function()
		RemoveCubbyESP(cubby)
	end))

	CubbyTracked[cubby] = { conns = conns }
end

local function RefreshCubbyESP()
	for cubby in pairs(CubbyTracked) do
		RemoveCubbyESP(cubby)
	end
	if CubbyMainConn then
		CubbyMainConn:Disconnect()
		CubbyMainConn = nil
	end
	if not Toggles.CubbyESPToggle or not Toggles.CubbyESPToggle.Value then return end

	local currentRooms = Workspace:FindFirstChild("CurrentRooms")
	if not currentRooms then return end

	for _, desc in ipairs(currentRooms:GetDescendants()) do
		if VALID_CUBBIES[desc.Name] then
			AddCubbyESP(desc)
		end
	end

	CubbyMainConn = currentRooms.DescendantAdded:Connect(function(desc)
		if Toggles.CubbyESPToggle and Toggles.CubbyESPToggle.Value and VALID_CUBBIES[desc.Name] then
			AddCubbyESP(desc)
			task.defer(function()
				SyncAllCubbiesInRoom(currentRooms)
			end)
		end
	end)
end

local function RefreshCategory(def)
	local isOn = Toggles[def.Toggle] and Toggles[def.Toggle].Value
	local col = (Options[def.Color] and Options[def.Color].Value) or def.DefaultColor

	if def.Toggle == "CubbyESPToggle" then
		RefreshCubbyESP()
		return
	end

	local list = ESPController.Objects[def.Cat] or {}
	for _, obj in ipairs(list) do
		if isOn and obj and obj.Parent then
			if def.Toggle == "DoorESPToggle" then
				if IsDupeOrVacuumDoor(obj) then
					ApplyObjectESP(obj, obj.Name == "Vacuum" and "Vacuum" or "Dupe", Color3.fromRGB(255, 0, 0), def.RoomBased, nil, "DupeVacuumRed")
				else
					ApplyObjectESP(obj, ResolveDoorLabel(obj), col, def.RoomBased, ResolveDoorLabel, "Highlight")
				end
			elseif def.Toggle == "PlayerESPToggle" then
				local plr = Players:GetPlayerFromCharacter(obj)
				if plr and plr ~= LocalPlayer then
					ApplyObjectESP(obj, plr.Name, col, false, function() return plr.Name end, "Highlight")
				end
			else
				local rawKey = obj:GetAttribute("ESPRawKey") or tostring(obj.Name)
				local st = obj:GetAttribute("ESPStyle") or def.Style or "Highlight"
				ApplyObjectESP(obj, rawKey, col, def.RoomBased, nil, st)
			end
		else
			ClearObjectESP(obj)
		end
	end
end

	end

local function HasObjectiveAncestor(inst)
	if not inst then return false end
	local cur = inst.Parent
	while cur and cur ~= Workspace do
		if ObjectiveNames[cur.Name] then

local function RegisterInList(cat, obj, rawKey, style, extraSuffix, keepWhenDropped)
	if not obj or not obj.Parent then return end
	pcall(function()
		obj:SetAttribute("ESPRawKey", rawKey or tostring(obj.Name))
		obj:SetAttribute("ESPStyle", style or "Highlight")
		obj:SetAttribute("ESPExtraSuffix", extraSuffix or "")
		if keepWhenDropped ~= nil then
			obj:SetAttribute("KeepWhenDropped", keepWhenDropped)
		end
	end)
	local arr = ESPController.Objects[cat]
	if arr and not table.find(arr, obj) then
		table.insert(arr, obj)
	end
end

local function IsValidTrashTarget(obj)
	if not obj or not obj.Name then return false end
	if not TrashNameSet[obj.Name] then return false end
	if IsHeldByLocalPlayer(obj) then return false end
	if IsInsideBookshelfOrDecor(obj) then return false end
	if obj:IsA("Model") or obj:IsA("BasePart") or obj:FindFirstChildWhichIsA("BasePart", true) then
		return true
	end
	return false
end

local function IsValidChestTarget(obj)
	if not obj or not obj.Name then return false end
	if ChestNames[obj.Name] then return true end
	local lower = string.lower(obj.Name)
	if string.find(lower, "chest", 1, true) or string.find(lower, "toolbox", 1, true) then
		if obj:IsA("Model") or obj:IsA("BasePart") or obj:FindFirstChildWhichIsA("BasePart", true) then
			return true
		end
	end
	return false
end

local function IsValidGoldTarget(obj)
	if not obj or not obj.Name then return false end
	if CurrencyNames[obj.Name] then return true end
	if string.find(string.lower(obj.Name), "goldpile", 1, true) or string.find(string.lower(obj.Name), "stardust", 1, true) then
		if obj:IsA("Model") or obj:IsA("BasePart") or obj:FindFirstChildWhichIsA("BasePart", true) then
			return true
		end
	end
	return false
end

local function ClassifyAndAttachESP(obj)
	if not obj or not obj.Name then return end
	local name = obj.Name

	local curRooms = Workspace:FindFirstChild("CurrentRooms")
	if curRooms then
		for _, room in ipairs(curRooms:GetChildren()) do
			local rNum = tonumber(room.Name)
			if rNum and obj:IsDescendantOf(room) then
				obj:SetAttribute("ParentRoom", rNum)
				break
			end
		end
	end

	if IsDupeOrVacuumDoor(obj) then
		local dvLabel = (name == "Vacuum") and "Vacuum" or "Dupe"
		RegisterInList("Doors", obj, dvLabel, "DupeVacuumRed", "")
		if Toggles.DoorESPToggle and Toggles.DoorESPToggle.Value then
			ApplyObjectESP(obj, dvLabel, Color3.fromRGB(255, 0, 0), true, nil, "DupeVacuumRed")
		end
		return
	end

	if name == "Door" and obj:IsA("Model") and obj.Parent and tonumber(obj.Parent.Name) and not IsElevatorOrBoardedDoor(obj) then
		RegisterInList("Doors", obj, "门", "Highlight", "")
		if Toggles.DoorESPToggle and Toggles.DoorESPToggle.Value then
			ApplyObjectESP(obj, ResolveDoorLabel(obj), Options.DoorESPColor.Value, true, ResolveDoorLabel, "Highlight")
		end
	end

	if HidingSpotNames[name] or string.find(string.lower(name), "hidingspot", 1, true) then
		RegisterInList("HidingSpots", obj, name, "Highlight", "")
		if Toggles.HidingSpotESPToggle and Toggles.HidingSpotESPToggle.Value then
			ApplyObjectESP(obj, name, Options.HidingSpotESPColor.Value, true, nil, "Highlight")
		end
	end

	if IsValidChestTarget(obj) then
		local isLocked = (obj:GetAttribute("Locked") == true) or (obj:FindFirstChild("Lock", true) ~= nil)
		local suffix = isLocked and "[Locked]" or ""
		RegisterInList("Chests", obj, name, "Highlight", suffix)
		if Toggles.ChestESPToggle and Toggles.ChestESPToggle.Value then
			ApplyObjectESP(obj, name, Options.ChestESPColor.Value, true, nil, "Highlight")
		end
	end

	if IsValidTrashTarget(obj) then
		RegisterInList("Trash", obj, name, "Highlight", "")
		if Toggles.TrashESPToggle and Toggles.TrashESPToggle.Value then
			ApplyObjectESP(obj, name, Options.TrashESPColor.Value, true, nil, "Highlight")
		end
	end

	if IsRealWorldItem(obj) then
		local itemCol = (Options.ItemESPColor and Options.ItemESPColor.Value) or Color3.fromRGB(190, 90, 255)
		RegisterInList("Items", obj, name, "ItemFullPurple", "", true)
		if Toggles.ItemESPToggle and Toggles.ItemESPToggle.Value then
			ApplyObjectESP(obj, name, itemCol, true, nil, "ItemFullPurple")
		end
	end

	if IsValidGoldTarget(obj) then
		local val = obj:GetAttribute("GoldValue") or (obj:FindFirstChild("GoldValue") and obj.GoldValue.Value)
		local suffix = val and ("[" .. tostring(val) .. "]") or ""
		local rawKey = CurrencyNames[name] and name or "GoldPile"
		RegisterInList("Currency", obj, rawKey, "Highlight", suffix)
		if Toggles.CurrencyESPToggle and Toggles.CurrencyESPToggle.Value then
			ApplyObjectESP(obj, rawKey, Options.CurrencyESPColor.Value, true, nil, "Highlight")
		end
	end

	if name == "Ladder" then
		RegisterInList("Ladders", obj, "Ladder", "Highlight", "")
		if Toggles.LadderESPToggle and Toggles.LadderESPToggle.Value then
			ApplyObjectESP(obj, "Ladder", Options.LadderESPColor.Value, true, nil, "Highlight")
		end
	end

	if MiscNames[name] then
		local isWhiteItem = WhiteOutlineMiscSet[name] == true
		local miscStyle = isWhiteItem and "WhiteOutline" or "Highlight"
		local miscCol = isWhiteItem and Color3.fromRGB(255, 255, 255) or Options.MiscESPColor.Value
		RegisterInList("Misc", obj, name, miscStyle, "")
		if Toggles.MiscESPToggle and Toggles.MiscESPToggle.Value then
			ApplyObjectESP(obj, name, miscCol, true, nil, miscStyle)
		end
	end

	if ObjectiveNames[name] and not HasObjectiveAncestor(obj) then
		local objCol = (Options.ObjectiveESPColor and Options.ObjectiveESPColor.Value) or Color3.fromRGB(0, 255, 0)
		RegisterInList("Objectives", obj, name, "Highlight", "")
		if Toggles.ObjectiveESPToggle and Toggles.ObjectiveESPToggle.Value then
			ApplyObjectESP(obj, name, objCol, true, nil, "Highlight")
		end
	end

	local entInfo = Entities[name]
	if entInfo or name == "FigureRig" or name == "GiggleCeiling" or name == "Snare" then
		local alias = entInfo and entInfo.Alias or name
		local style = ResolveEntityVisualStyle(alias)
		local isAlwaysRed = (style == "SquareBox" or style == "RedOutlineShape" or style == "DupeVacuumRed" or style == "EyesScreechRing")
		local col = isAlwaysRed and Color3.fromRGB(255, 0, 0) or Options.EntityESPColor.Value
		RegisterInList("Entities", obj, alias, style, "")
		if Toggles.EntityESPToggle and Toggles.EntityESPToggle.Value then
			local optVal = Options.EntityESPOptions and Options.EntityESPOptions.Value
			local allowed = true
			if type(optVal) == "table" then
				local anyChecked = false
				for _, v in pairs(optVal) do if v then anyChecked = true break end end
				if anyChecked then allowed = (optVal[alias] == true) end
			end
			if allowed then
				ApplyObjectESP(obj, alias, col, false, nil, style)
			end
		end
	end

	if VALID_CUBBIES[name] and IsInsideRoom(obj) then
		RegisterInList("Cubbies", obj, name, "Highlight", "")
		AddCubbyESP(obj)
	end
end

local function HookPlayerESP(plr)
	if not plr or plr == LocalPlayer then return end
	local function attachChar(char)
		if not char then return end
		RegisterInList("Players", char, plr.Name, "Highlight", "")
		if Toggles.PlayerESPToggle and Toggles.PlayerESPToggle.Value then
			ApplyObjectESP(char, plr.Name, Options.PlayerESPColor.Value, false, function() return plr.Name end, "Highlight")
		end
	end
	if plr.Character then attachChar(plr.Character) end
	plr.CharacterAdded:Connect(attachChar)
end

if Players then
	for _, plr in ipairs(Players:GetPlayers()) do
		HookPlayerESP(plr)
	end
	Players.PlayerAdded:Connect(HookPlayerESP)
end

if Workspace then
	for _, desc in ipairs(Workspace:GetDescendants()) do
		ClassifyAndAttachESP(desc)
	end
	Workspace.DescendantAdded:Connect(function(desc)
		task.defer(function()
			ClassifyAndAttachESP(desc)
		end)
	end)
end
