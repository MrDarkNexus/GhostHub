local function WatchEntitiesWithSound()
	local liveEntities = Workspace:FindFirstChild("LiveEntities")
	if liveEntities then
		for _, child in ipairs(liveEntities:GetChildren()) do
			if not NotifiedEntities[child] then
				NotifiedEntities[child] = true
				NotifyEntitySpawn(child.Name)
			end
		end
		liveEntities.ChildAdded:Connect(function(child)
			if not NotifiedEntities[child] then
				NotifiedEntities[child] = true
				NotifyEntitySpawn(child.Name)
			end
		end)
	end
end

	local function TPDropsToNearestGrinder()
		local drops = Workspace:FindFirstChild("Drops")
		local rooms = Workspace:FindFirstChild("CurrentRooms")
		local char = LocalPlayer.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if not (drops and rooms and hrp) then return end

		local nearestGrinder, bestDist = nil, math.huge
		for _, room in ipairs(rooms:GetChildren()) do
			local scrapper = room:FindFirstChild("StairwellScrapper", true)
			if scrapper then
				for _, obj in ipairs(scrapper:GetDescendants()) do
					if obj.Name == "Grinder1" and (obj:IsA("Model") or obj:IsA("BasePart")) then
						local d = (hrp.Position - obj:GetPivot().Position).Magnitude
						if d < bestDist then
							bestDist = d
							nearestGrinder = obj
						end
					end
				end
			end
		end
		if nearestGrinder then
			local cf = nearestGrinder:GetPivot()
			for _, drop in ipairs(drops:GetChildren()) do
				if drop:IsA("Model") or drop:IsA("BasePart") then
					drop:PivotTo(cf)
				end
			end
		end
	end

	task.spawn(function()
		while ScreenGui.Parent do
			if XKState.BypassNoise then
				local misc = Workspace:FindFirstChild("Misc")
				if misc then
					for _, child in ipairs(misc:GetChildren()) do
						if child:IsA("Model") and child.Name == "TV_Stand" then
							local cf = child:GetPivot()
							if cf.Position.Y > -119 then
								child:PivotTo(CFrame.new(cf.Position.X, -120, cf.Position.Z))
							end
						end
					end
				end
			end

			if XKState.FlingCreak then
				local creak = Workspace:FindFirstChild("LiveEntities") and Workspace.LiveEntities:FindFirstChild("Creak")
				local misc = Workspace:FindFirstChild("Misc")
				if creak and misc then
					local targetCf = creak:GetPivot()
					for _, model in ipairs(misc:GetChildren()) do
						if model.Name == "ShoppingCart" and model:IsA("Model") then
							model:PivotTo(targetCf)
						end
					end
				end
			end

			if XKState.DisableStairwellCrusherCollision then
				local rooms = Workspace:FindFirstChild("CurrentRooms")
				if rooms then
					for _, room in ipairs(rooms:GetChildren()) do
						local assets = room:FindFirstChild("Assets")
						if assets then
							for _, container in ipairs(assets:GetChildren()) do
								if container.Name == "StairwellCrusherContainer1" then
									for _, desc in ipairs(container:GetDescendants()) do
										if desc:IsA("BasePart") and desc.CanCollide then
											desc.CanCollide = false
										end
									end
								end
							end
						end
					end
				end
			end

			if XKState.MeldRemover then
				local rooms = Workspace:FindFirstChild("CurrentRooms")
				if rooms then
					for _, room in ipairs(rooms:GetChildren()) do
						for _, folderName in ipairs({ "Assets", "Parts" }) do
							local folder = room:FindFirstChild(folderName)
							if folder then
								for _, child in ipairs(folder:GetChildren()) do
									if child.Name == "MeldWall" or child.Name == "Boleahghth29tdgfhy2thuy2htuu259uhh3u" then
										child:Destroy()
									end
								end
							end
						end
					end
				end
			end

			if XKState.EnableDroppedItemsInterval then
				BringDroppedItems()
				task.wait(math.max(0.05, XKState.DroppedItemsInterval))
			else
				task.wait(0.25)
			end
		end
	end)

	local quickActions = {
		{ text = "拾取掉落物 (Bring Drops)", fn = BringDroppedItems },
		{ text = "传送所有掉落物到最近的研磨机", fn = TPDropsToNearestGrinder },
		{ text = "Test Entity Bildirim Sesi", fn = function() NotifyEntitySpawn("TestEntity") end },
	}

	local yPos = 14
	for _, act in ipairs(quickActions) do
		local b = Instance.new("TextButton")
		b.Parent = mainPage
		b.Size = UDim2.new(0, 320, 0, 34)
		b.Position = UDim2.new(0, 14, 0, yPos)
		b.BackgroundColor3 = Color3.fromRGB(42, 48, 70)
		b.Font = Enum.Font.Code
		b.TextSize = 12
		b.TextColor3 = Color3.fromRGB(255, 255, 255)
		b.Text = act.text
		b.MouseButton1Click:Connect(act.fn)
		Instance.new("UICorner", b).CornerRadius = UDim.new(0, 6)
		yPos += 42
	end
end
