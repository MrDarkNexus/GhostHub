local function SafeLoad(Url, Label)
	local tag = tostring(Label or Url)

	if type(Url) ~= "string" or #Url == 0 then
		warn("[SafeLoad] Geçersiz URL (" .. tag .. ")")
		return nil
	end

	local okFetch, sourceOrErr = pcall(function()
		return game:HttpGet(Url)
	end)
	if not okFetch or type(sourceOrErr) ~= "string" or #sourceOrErr == 0 then
		warn("[SafeLoad] İndirme başarısız (" .. tag .. "): " .. tostring(sourceOrErr))
		return nil
	end

	local fn, compileErr = loadstring(sourceOrErr, "@" .. tag)
	if not fn then
		warn("[SafeLoad] Derleme hatası (" .. tag .. "): " .. tostring(compileErr))
		return nil
	end

	local okRun, result = pcall(fn)
	if not okRun then
		warn("[SafeLoad] Çalıştırma hatası (" .. tag .. "): " .. tostring(result))
		return nil
	end

	return result
end

	if Abysall and Abysall.Environment and type(Abysall.Environment.cloneref) == "function" then
		local ok, res = pcall(Abysall.Environment.cloneref, Object)

if not Abysall.ESPLibrary then
	local FallbackESP = {
		Objects = {},
		ColorTable = {},
		Rainbow = false,
		ShowDistance = true,
		FillTransparency = 0.75,
		OutlineTransparency = 0,
		TextTransparency = 0,
		TextOutlineTransparency = 0,
		TextSize = 20,
		Font = Enum.Font.Highway,
		RenderLimit = 240,
		FadeTime = 0.25,
	}

	function FallbackESP:GenerateRandomString()
		return "XK_ESP_" .. tostring(math.random(100000, 999999))
	end

	local function getAdornee(obj)
		if typeof(obj) ~= "Instance" then return nil end
		if obj:IsA("BasePart") then return obj end
		if obj:IsA("Model") then
			return obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)
		end
		return obj:FindFirstChildWhichIsA("BasePart", true)
	end

	function FallbackESP:AddESP(opts)
		if type(opts) ~= "table" or not opts.Object then return end
		local obj = opts.Object
		self:RemoveESP(obj)

		local color = opts.Color or Color3.fromRGB(255, 255, 255)
		self.ColorTable[obj] = color

		local hl = Instance.new("Highlight")
		hl.Name = "XK_ESP_Highlight"
		hl.Adornee = obj
		hl.FillColor = color
		hl.OutlineColor = color
		hl.FillTransparency = self.FillTransparency
		hl.OutlineTransparency = self.OutlineTransparency
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.Parent = CoreGui

		local adornee = getAdornee(obj)
		local bb, lbl
		if adornee then
			bb = Instance.new("BillboardGui")
			bb.Name = "XK_ESP_Billboard"
			bb.Adornee = adornee
			bb.Size = UDim2.new(0, 220, 0, 48)
			bb.StudsOffset = Vector3.new(0, 2.2, 0)
			bb.AlwaysOnTop = true
			bb.MaxDistance = self.RenderLimit or 240
			bb.Parent = CoreGui

			lbl = Instance.new("TextLabel")
			lbl.Size = UDim2.new(1, 0, 1, 0)
			lbl.BackgroundTransparency = 1
			lbl.Text = tostring(opts.Text or obj.Name)
			lbl.TextColor3 = color
			lbl.TextStrokeColor3 = Color3.new(0, 0, 0)
			lbl.TextStrokeTransparency = self.TextOutlineTransparency
			lbl.TextTransparency = self.TextTransparency
			lbl.TextSize = self.TextSize
			lbl.Font = self.Font
			lbl.Parent = bb
		end

		self.Objects[obj] = {
			Options = opts,
			Highlight = hl,
			Billboard = bb,
			Label = lbl,
			Adornee = adornee,
			BaseText = tostring(opts.Text or obj.Name),
		}
	end

	function FallbackESP:UpdateObjectColor(obj, color)
		if not obj or not color then return end
		self.ColorTable[obj] = color
		local entry = self.Objects[obj]
		if entry then
			entry.Options.Color = color
			if entry.Highlight then
				entry.Highlight.FillColor = color
				entry.Highlight.OutlineColor = color
			end
			if entry.Label then
				entry.Label.TextColor3 = color
			end
		end
	end

	function FallbackESP:RemoveESP(obj)
		if not obj then return end
		local entry = self.Objects[obj]
		if entry then
			if entry.Highlight then pcall(function() entry.Highlight:Destroy() end) end
			if entry.Billboard then pcall(function() entry.Billboard:Destroy() end) end
			self.Objects[obj] = nil
		end
	end

	function FallbackESP:SetRainbow(v) self.Rainbow = v end
	function FallbackESP:SetShowDistance(v) self.ShowDistance = v end
	function FallbackESP:SetFillTransparency(v)
		self.FillTransparency = v
		for _, e in pairs(self.Objects) do if e.Highlight then e.Highlight.FillTransparency = v end end
	end
	function FallbackESP:SetOutlineTransparency(v)
		self.OutlineTransparency = v
		for _, e in pairs(self.Objects) do if e.Highlight then e.Highlight.OutlineTransparency = v end end
	end
	function FallbackESP:SetTextTransparency(v)
		self.TextTransparency = v
		for _, e in pairs(self.Objects) do if e.Label then e.Label.TextTransparency = v end end
	end
	function FallbackESP:SetTextOutlineTransparency(v)
		self.TextOutlineTransparency = v
		for _, e in pairs(self.Objects) do if e.Label then e.Label.TextStrokeTransparency = v end end
	end
	function FallbackESP:SetFadeTime(v) self.FadeTime = v end
	function FallbackESP:SetRenderLimit(v)
		self.RenderLimit = v
		for _, e in pairs(self.Objects) do if e.Billboard then e.Billboard.MaxDistance = v end end
	end
	function FallbackESP:SetTextSize(v)
		self.TextSize = v
		for _, e in pairs(self.Objects) do if e.Label then e.Label.TextSize = v end end
	end
	function FallbackESP:SetFont(f)
		self.Font = f
		for _, e in pairs(self.Objects) do if e.Label then e.Label.Font = f end end
	end
	function FallbackESP:SetTracers(_) end
	function FallbackESP:SetTracerSize(_) end
	function FallbackESP:SetTracerOrigin(_) end
	function FallbackESP:SetArrows(_) end
	function FallbackESP:SetArrowRadius(_) end
	function FallbackESP:SetDistanceSizeRatio(_) end
	function FallbackESP:Unload()
		for obj in pairs(self.Objects) do
			self:RemoveESP(obj)
		end
	end

	RunService.Heartbeat:Connect(function()
		local char = LocalPlayer and LocalPlayer.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hue = (tick() * 0.25) % 1
		local rainbowCol = Color3.fromHSV(hue, 1, 1)

		for obj, entry in pairs(FallbackESP.Objects) do
			if not obj or not obj.Parent then
				FallbackESP:RemoveESP(obj)
			else
				local col = FallbackESP.Rainbow and rainbowCol or (entry.Options.Color or Color3.fromRGB(255, 255, 255))
				if entry.Highlight then
					entry.Highlight.FillColor = col
					entry.Highlight.OutlineColor = col
					entry.Highlight.FillTransparency = FallbackESP.FillTransparency
					entry.Highlight.OutlineTransparency = FallbackESP.OutlineTransparency
				end
				if entry.Label then
					entry.Label.TextColor3 = col
					entry.Label.TextTransparency = FallbackESP.TextTransparency
					entry.Label.TextStrokeTransparency = FallbackESP.TextOutlineTransparency
					entry.Label.TextSize = FallbackESP.TextSize
					entry.Label.Font = FallbackESP.Font

					local distStr = ""
					if FallbackESP.ShowDistance and hrp and entry.Adornee and entry.Adornee.Parent then
						local d = math.floor((hrp.Position - entry.Adornee.Position).Magnitude)
						distStr = " [" .. d .. "]"
					end
					entry.Label.Text = tostring(entry.Options.Text or entry.BaseText) .. distStr
				end
			end
		end
	end)

	Abysall.ESPLibrary = FallbackESP

local CurrentLang = "zh"
local TranslatableElements = {}

local function GetTranslatedText(originalText, lang)
	if lang == "zh" or type(originalText) ~= "string" then
		return originalText
	end
	local entry = Translations[originalText]
	if entry and entry[lang] then
		return entry[lang]
	end
	return originalText
end

local function RegisterTranslatable(setFn, getFn)
	table.insert(TranslatableElements, { set = setFn, get = getFn })
end

local function ApplyLanguage(lang)
	CurrentLang = (lang == "tr" or lang == "en" or lang == "zh") and lang or "tr"

	for _, item in ipairs(TranslatableElements) do
		pcall(function()
			local orig = item.get()
			if type(orig) == "string" then
				item.set(GetTranslatedText(orig, CurrentLang))
			end
		end)
	end

	local root = UILibrary.ScreenGui or CoreGui
	if root then
		for _, desc in ipairs(root:GetDescendants()) do
			if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
				if not desc:GetAttribute("OrigText") then
					desc:SetAttribute("OrigText", desc.Text)
				end
				local origText = desc:GetAttribute("OrigText")
				local mapped = GetTranslatedText(origText, CurrentLang)
				if mapped ~= origText then
					desc.Text = mapped
				elseif CurrentLang == "zh" then
					desc.Text = origText
				end
			end
		end
	end
end

local function WrapTextContainer(container, origText)
	if not container then return end
	if type(container.SetText) == "function" then
		local current = origText
		RegisterTranslatable(
			function(txt) container:SetText(txt) end,
			function() return current end
		)
	end
end

local function IsCameraFirstPerson(cam)
	if not cam then return false end

local function NeutralizeCameraShake(cam, char)
	if not NoCameraShakeEnabled then return end
	pcall(function()
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.CameraOffset = Vector3.zero
		end
		if cam then
			local shakeVal = cam:FindFirstChild("CameraShake") or cam:FindFirstChild("Shake")
			if shakeVal and (shakeVal:IsA("NumberValue") or shakeVal:IsA("Vector3Value")) then
				if shakeVal:IsA("NumberValue") then
					shakeVal.Value = 0
				elseif shakeVal:IsA("Vector3Value") then
					shakeVal.Value = Vector3.zero
				end
			end
		end
	end)
end

local function DetectCurrentFloorKey()
	local floorName = ""
	pcall(function()
		local rs = ReplicatedStorage
		local gd = rs and rs:FindFirstChild("GameData")
		local fl = gd and gd:FindFirstChild("Floor")
		if fl and fl.Value then
			floorName = tostring(fl.Value)
		end
	end)
	local lower = string.lower(floorName)
	if string.find(lower, "fools", 1, true) or string.find(lower, "april", 1, true) then
		return "AprilFools"
	elseif string.find(lower, "cringle", 1, true) or string.find(lower, "workshop", 1, true) or string.find(lower, "frozen", 1, true) then
		return "CringleWorkshop"
	elseif string.find(lower, "archive", 1, true) then
		return "Archives"
	elseif string.find(lower, "stair", 1, true) then
		return "Stairwell"
	elseif string.find(lower, "outdoor", 1, true) then
		return "OutDoors"
	elseif string.find(lower, "backdoor", 1, true) then
		return "BackDoors"
	elseif string.find(lower, "mine", 1, true) then
		return "Mines"
	elseif string.find(lower, "hotel", 1, true) then
		return "Hotel"
	end
	return nil
end

local function ResolveAliasForCurrentMode(rawKey)
	local entry = Entities[rawKey]
	local alias = entry and entry.Alias or tostring(rawKey)
	local floorKey = DetectCurrentFloorKey()

	if not floorKey then
		return "CustomEntity"
	end

	local allowed = FloorAllowedAliases[floorKey]
	if allowed and not allowed[alias] then
		return "CustomEntity"
	end

	return alias
end

local function IsInsideRoom(inst)
	if not inst or not Workspace then return false end
	local currentRooms = Workspace:FindFirstChild("CurrentRooms")
	if not currentRooms then return false end
	return inst:IsDescendantOf(currentRooms)
end

local function IsEntitySelected(aliasName)
	local dd = Options and Options.EntityList
	if not dd or type(dd.Value) ~= "table" then
		return true
	end
	local hasSelected = false
	for _, isOn in pairs(dd.Value) do
		if isOn then
			hasSelected = true
			break
		end
	end
	if not hasSelected then
		return true
	end
	return dd.Value[aliasName] == true or (dd.Value["CustomEntity"] == true and aliasName == "CustomEntity")
end

local function TriggerEntityNotify(keyOrAlias)
	local resolvedAlias = ResolveAliasForCurrentMode(keyOrAlias)
	if not IsEntitySelected(resolvedAlias) then return end

	PlayEntitySpawnSound()
	ShowWhiteAlertBanner('"' .. resolvedAlias .. '" Spawned', 3.5)

	local customInput = Options and Options.EntityChatMessage and Options.EntityChatMessage.Value
	local notifyLabel = (type(customInput) == "string" and #customInput > 0 and customInput ~= "已生成！")
		and (resolvedAlias .. " - " .. customInput)
		or (resolvedAlias .. " Spawned")

	pcall(function()
		Library:Notify('⚠️ "' .. notifyLabel .. '"', 4)
	end)
end

local function IsHeldByLocalPlayer(inst)
	if not inst then return false end
	local char = LocalPlayer and LocalPlayer.Character
	local backpack = LocalPlayer and LocalPlayer:FindFirstChildOfClass("Backpack")
	if char and inst:IsDescendantOf(char) then return true end
	if backpack and inst:IsDescendantOf(backpack) then return true end
	return false
end

local function IsItemSelected(itemName)
	local dd = Options and Options.NotifyItemList
	if not dd or type(dd.Value) ~= "table" then
		return false
	end
	return dd.Value[itemName] == true
end

local function TriggerItemNotify(itemName)
	if not ItemNameSet[itemName] then return end
	if not IsItemSelected(itemName) then return end

	PlayEntitySpawnSound()
	ShowWhiteAlertBanner('"' .. itemName .. '" Spawned', 3.2)

	pcall(function()
		Library:Notify('⚠️ "' .. itemName .. '" Spawned', 4)
	end)
end

local SeenDebugIds = {}
local function MarkSeen(inst)
	if not inst then return false end
	local ok, id = pcall(function() return inst:GetDebugId() end)
	local key = ok and tostring(id) or tostring(inst)
	if SeenDebugIds[key] then
		return false
	end
	SeenDebugIds[key] = true
	task.delay(6, function()
		SeenDebugIds[key] = nil
	end)
	return true
end

local function OnWorldDescendantAdded(desc)
	if not desc or not desc.Name then return end
	local nameStr = tostring(desc.Name)

	local matchedEntityKey = Entities[nameStr] and nameStr or nil
	if not matchedEntityKey then
		local lowerName = string.lower(nameStr)
		for k in pairs(Entities) do
			if string.find(lowerName, string.lower(k), 1, true) then
				matchedEntityKey = k
				break
			end
		end
	end
	if matchedEntityKey and IsInsideRoom(desc) and MarkSeen(desc) then
		TriggerEntityNotify(matchedEntityKey)
		return
	end

	if ItemNameSet[nameStr] and IsInsideRoom(desc) and not IsHeldByLocalPlayer(desc) and MarkSeen(desc) then
		TriggerItemNotify(nameStr)
	end
end

if Workspace then
	Workspace.DescendantAdded:Connect(OnWorldDescendantAdded)
	for _, obj in ipairs(Workspace:GetDescendants()) do
		OnWorldDescendantAdded(obj)
	end
end

				for _, found in ipairs(currentRooms:GetDescendants()) do
					if found.Name == itemName and not IsHeldByLocalPlayer(found) and MarkSeen(found) then
						TriggerItemNotify(itemName)
					end
				end
			end
		end
	end
end)

local function GetFloorValue()
	local fl = ""
	pcall(function()
		local gd = ReplicatedStorage:FindFirstChild("GameData")
		local fObj = gd and gd:FindFirstChild("Floor")
		if fObj then fl = tostring(fObj.Value) end
	end)
	return fl
end

-- Kapı numarasını bozmadan altına [Locked] veya [Opened] ekler
local function ResolveDoorLabel(doorModel)
	local floor = GetFloorValue()
	local num = tonumber(doorModel.Parent and doorModel.Parent.Name)
		or tonumber(doorModel.Parent and doorModel.Parent.Parent and doorModel.Parent.Parent.Name)

	if num then
		num = num + 1
		if floor == "Mines" then
			num = num + 100
		elseif floor == "Backdoor" then
			num = num - 50
		end
	end

	local baseTitle = num and ("Door " .. tostring(num)) or tostring(doorModel.Name or "Door")

	local isOpened = (doorModel:GetAttribute("Opened") == true) or (doorModel:GetAttribute("Open") == true)
	local isLocked = (doorModel:GetAttribute("Locked") == true) or (doorModel:FindFirstChild("Lock", true) ~= nil)

	local statusTag = nil
	if isOpened then
		statusTag = "[Opened]"
	elseif isLocked then
		statusTag = "[Locked]"
	end

	if statusTag then
		return baseTitle .. "\n" .. statusTag
	end
	return baseTitle
end

local function IsRoomVisible(obj, roomBased)
	if not roomBased then return true end
	local curRoom = tonumber(LocalPlayer and LocalPlayer:GetAttribute("CurrentRoom"))
	local objRoom = tonumber(obj and obj:GetAttribute("ParentRoom"))
	if not curRoom or not objRoom then return true end
	if obj and obj:GetAttribute("IsDoorHighlight") then
		return objRoom == curRoom or objRoom == curRoom + 1
	end
	return objRoom == curRoom
end

local function ClearObjectESP(obj)
	if not obj then return end
	if ESPController.RoomConnections[obj] then
		for _, conn in ipairs(ESPController.RoomConnections[obj]) do
			pcall(function() conn:Disconnect() end)
		end
		ESPController.RoomConnections[obj] = nil
	end
	ESPController.ActiveESP[obj] = nil
	pcall(function()
		Abysall.ESPLibrary:RemoveESP(obj)
	end)
end

local function ApplyObjectESP(obj, text, color, roomBased, dynamicTextFn)
	if not obj or not obj.Parent then return end
	ClearObjectESP(obj)

	local function refresh()
		if not obj or not obj.Parent then
			ClearObjectESP(obj)
			return
		end
		local label = type(dynamicTextFn) == "function" and dynamicTextFn(obj) or text
		if not label then
			pcall(function() Abysall.ESPLibrary:RemoveESP(obj) end)
			return
		end
		if IsRoomVisible(obj, roomBased) then
			pcall(function()
				Abysall.ESPLibrary:AddESP({
					Object = obj,
					Text = label,
					Color = color,
				})
			end)
		else
			pcall(function() Abysall.ESPLibrary:RemoveESP(obj) end)
		end
	end

	ESPController.ActiveESP[obj] = {
		Text = text,
		Color = color,
		RoomBased = roomBased,
		DynamicTextFn = dynamicTextFn,
		Refresh = refresh,
	}

	refresh()

	local conns = {}
	if roomBased and LocalPlayer then
		table.insert(conns, LocalPlayer:GetAttributeChangedSignal("CurrentRoom"):Connect(refresh))
	end
	table.insert(conns, obj:GetAttributeChangedSignal("Opened"):Connect(refresh))
	table.insert(conns, obj:GetAttributeChangedSignal("Open"):Connect(refresh))
	table.insert(conns, obj:GetAttributeChangedSignal("Locked"):Connect(refresh))
	table.insert(conns, obj.DescendantAdded:Connect(function(d)
		if d.Name == "Lock" then
			refresh()
		end
	end))
	table.insert(conns, obj.Destroying:Once(function()
		ClearObjectESP(obj)
	end))
	ESPController.RoomConnections[obj] = conns
end

local function UpdateCategoryColor(categoryName, newColor)
	local list = ESPController.Objects[categoryName]
	if not list then return end
	for _, obj in ipairs(list) do
		if obj and obj.Parent and ESPController.ActiveESP[obj] then
			ESPController.ActiveESP[obj].Color = newColor
			pcall(function()
				Abysall.ESPLibrary:UpdateObjectColor(obj, newColor)
			end)
		end
	end
end

-- // Cubby ESP Sistemi (Cubby.lua tabanlı)
local VALID_CUBBIES = {
	Cubby_Large = true,
	Cubby_Tall = true,
	Cubby_Small = true,
	Cubby_Medium = true,
}
local CubbyTracked = {}
local CubbyMainConn = nil

local function IsCubbyLooted(cubby)
	if not cubby or not cubby.Parent then return true end
	local lootables = {}
	for _, child in ipairs(cubby:GetChildren()) do
		if child.Name == "LootableCubby" then
			table.insert(lootables, child)
		end
	end
	if #lootables == 0 then
		return false
	end
	local opened = 0
	for _, lootable in ipairs(lootables) do
		if lootable:FindFirstChild("LootHolder", true) then
			opened += 1
		end
	end
	return opened >= #lootables
end

local function RemoveCubbyESP(cubby)
	local data = CubbyTracked[cubby]
	if data then
		for _, c in ipairs(data.conns or {}) do
			pcall(function() c:Disconnect() end)
		end
		CubbyTracked[cubby] = nil
	end
	ClearObjectESP(cubby)
end

local function AddCubbyESP(cubby)
	if not Toggles.CubbyESPToggle or not Toggles.CubbyESPToggle.Value then return end
	if not VALID_CUBBIES[cubby.Name] then return end
	if IsCubbyLooted(cubby) then
		RemoveCubbyESP(cubby)
		return
	end
	if CubbyTracked[cubby] then return end

	local col = (Options.CubbyESPColor and Options.CubbyESPColor.Value) or Color3.fromRGB(255, 105, 180)
	ApplyObjectESP(cubby, cubby.Name, col, true)

	local conns = {}
	local function checkState()
		if IsCubbyLooted(cubby) then
			RemoveCubbyESP(cubby)
		end
	end
	table.insert(conns, cubby.DescendantAdded:Connect(function(d)
		if d.Name == "LootHolder" or d.Name == "LootableCubby" then
			task.defer(checkState)
		end
	end))
	table.insert(conns, cubby.DescendantRemoving:Connect(function(d)
		if d.Name == "LootHolder" or d.Name == "LootableCubby" then
			task.defer(checkState)
		end
	end))
	table.insert(conns, cubby.Destroying:Once(function()
		RemoveCubbyESP(cubby)
	end))

	CubbyTracked[cubby] = { conns = conns }
end

local function RefreshCubbyESP()
	for cubby in pairs(CubbyTracked) do
		RemoveCubbyESP(cubby)
	end
	if CubbyMainConn then
		CubbyMainConn:Disconnect()
		CubbyMainConn = nil
	end
	if not Toggles.CubbyESPToggle or not Toggles.CubbyESPToggle.Value then return end

	local currentRooms = Workspace:FindFirstChild("CurrentRooms")
	if not currentRooms then return end

	for _, desc in ipairs(currentRooms:GetDescendants()) do
		if VALID_CUBBIES[desc.Name] then
			AddCubbyESP(desc)
		end
	end

	CubbyMainConn = currentRooms.DescendantAdded:Connect(function(desc)
		if Toggles.CubbyESPToggle and Toggles.CubbyESPToggle.Value and VALID_CUBBIES[desc.Name] then
			AddCubbyESP(desc)
		end
	end)
end

	if not obj or not obj.Parent then return end
	if label then
		pcall(function() obj:SetAttribute("ESPText", label) end)

local function ClassifyAndAttachESP(obj)
	if not obj or not obj.Name then return end
	local name = obj.Name

	-- ParentRoom atama
	local curRooms = Workspace:FindFirstChild("CurrentRooms")
	if curRooms and not obj:GetAttribute("ParentRoom") then
		for _, room in ipairs(curRooms:GetChildren()) do
			local rNum = tonumber(room.Name)
			if rNum and obj:IsDescendantOf(room) then
				obj:SetAttribute("ParentRoom", rNum)
				break
			end
		end
	end

	-- 1. Door ESP (Kapı numarasını bozmadan [Locked] / [Opened] desteği)
	if name == "Door" and obj:IsA("Model") and obj.Parent and tonumber(obj.Parent.Name) then
		RegisterInList("Doors", obj, ResolveDoorLabel(obj))
		if Toggles.DoorESPToggle and Toggles.DoorESPToggle.Value then
			ApplyObjectESP(obj, ResolveDoorLabel(obj), Options.DoorESPColor.Value, true, ResolveDoorLabel)
		end
	end

	-- 2. HidingSpot ESP
	if HidingSpotNames[name] or string.find(string.lower(name), "hidingspot", 1, true) then
		RegisterInList("HidingSpots", obj, name)
		if Toggles.HidingSpotESPToggle and Toggles.HidingSpotESPToggle.Value then
			ApplyObjectESP(obj, name, Options.HidingSpotESPColor.Value, true)
		end
	end

	-- 3. Chest ESP
	if ChestNames[name] then
		local lbl = obj:GetAttribute("Locked") and (name .. " [Locked]") or name
		RegisterInList("Chests", obj, lbl)
		if Toggles.ChestESPToggle and Toggles.ChestESPToggle.Value then
			ApplyObjectESP(obj, lbl, Options.ChestESPColor.Value, true)
		end
	end

	-- 4. Item ESP
	if ItemNameSet[name] and not IsHeldByLocalPlayer(obj) then
		RegisterInList("Items", obj, name)
		if Toggles.ItemESPToggle and Toggles.ItemESPToggle.Value then
			ApplyObjectESP(obj, name, Options.ItemESPColor.Value, true)
		end
	end

	-- 5. Currency ESP
	if CurrencyNames[name] then
		local val = obj:GetAttribute("GoldValue")
		local lbl = val and (name .. " [" .. tostring(val) .. "]") or name
		RegisterInList("Currency", obj, lbl)
		if Toggles.CurrencyESPToggle and Toggles.CurrencyESPToggle.Value then
			ApplyObjectESP(obj, lbl, Options.CurrencyESPColor.Value, true)
		end
	end

	-- 6. Ladder ESP
	if name == "Ladder" then
		RegisterInList("Ladders", obj, "Ladder")
		if Toggles.LadderESPToggle and Toggles.LadderESPToggle.Value then
			ApplyObjectESP(obj, "Ladder", Options.LadderESPColor.Value, true)
		end
	end

	-- 7. Misc ESP
	if MiscNames[name] then
		RegisterInList("Misc", obj, name)
		if Toggles.MiscESPToggle and Toggles.MiscESPToggle.Value then
			ApplyObjectESP(obj, name, Options.MiscESPColor.Value, true)
		end
	end

	-- 8. Objective ESP
	if ObjectiveNames[name] then
		RegisterInList("Objectives", obj, name)
		if Toggles.ObjectiveESPToggle and Toggles.ObjectiveESPToggle.Value then
			ApplyObjectESP(obj, name, Options.ObjectiveESPColor.Value, true)
		end
	end

	-- 9. Entity ESP
	local entInfo = Entities[name]
	if entInfo or name == "FigureRig" or name == "GiggleCeiling" or name == "Snare" or name == "DoorFake" or name == "FakeDoor" then
		local alias = entInfo and entInfo.Alias or name
		RegisterInList("Entities", obj, alias)
		if Toggles.EntityESPToggle and Toggles.EntityESPToggle.Value then
			local optVal = Options.EntityESPOptions and Options.EntityESPOptions.Value
			local allowed = true
			if type(optVal) == "table" then
				local anyChecked = false
				for _, v in pairs(optVal) do if v then anyChecked = true break end end
				if anyChecked then allowed = (optVal[alias] == true) end
			end
			if allowed then
				ApplyObjectESP(obj, alias, Options.EntityESPColor.Value, false)
			end
		end
	end

	-- 10. Cubby ESP
	if VALID_CUBBIES[name] and IsInsideRoom(obj) then
		RegisterInList("Cubbies", obj, name)
		AddCubbyESP(obj)
	end
end

Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
	ActiveCamera = Workspace.CurrentCamera
	if ActiveCamera and not POVEnabled then
		OriginalPOV = ActiveCamera.FieldOfView
	end
end)

pcall(function()
	RunService:UnbindFromRenderStep("XK_VisualController")
end)

RunService:BindToRenderStep(
	"XK_VisualController",
	Enum.RenderPriority.Camera.Value + 1,
	function()
		if FullBrightEnabled then
			ApplyFullBrightState(true)
		end

		local cam = Workspace.CurrentCamera
		local char = LocalPlayer and LocalPlayer.Character
		if not cam then return end

		if NoCameraShakeEnabled then
			NeutralizeCameraShake(cam, char)
		end

		if POVEnabled then
			cam.FieldOfView = tonumber(POVValue) or 90
		end

		if ThirdPersonEnabled and char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local root = char:FindFirstChild("HumanoidRootPart")
			if hum and root and IsCameraFirstPerson(cam) then
				cam.CFrame = cam.CFrame * CFrame.new(0, 0, ThirdPersonDistance)
			end
		end
	end
)
